"use client";

import { useState, type FormEvent, Suspense } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import Link from "next/link";
import { Button, Input, EmptyState } from "@saachi/ui";

function ResetPasswordForm() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const token = searchParams.get("token");

  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [done, setDone] = useState(false);

  function handleSubmit(e: FormEvent) {
    e.preventDefault();
    if (password !== confirmPassword) return;
    setSubmitting(true);
    // NOTE: demo-only, no backend password-reset-confirm mutation wired up yet.
    setTimeout(() => {
      setSubmitting(false);
      setDone(true);
    }, 500);
  }

  if (!token) {
    return (
      <EmptyState
        title="Invalid or expired link"
        description="This password reset link is missing or no longer valid."
        action={
          <Link href="/forgot-password">
            <Button>Request a new link</Button>
          </Link>
        }
      />
    );
  }

  if (done) {
    return (
      <EmptyState
        title="Password updated"
        description="Your password has been reset. You can now log in."
        action={<Button onClick={() => router.push("/login")}>Go to login</Button>}
      />
    );
  }

  return (
    <>
      <h1 className="mb-1 font-display text-[26px] font-bold text-ink">Reset password</h1>
      <p className="mb-8 text-[14px] text-ink-soft">Choose a new password for your account.</p>

      <form onSubmit={handleSubmit} className="flex flex-col gap-4">
        <Input
          label="New password"
          type="password"
          required
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          placeholder="At least 8 characters"
        />
        <Input
          label="Confirm new password"
          type="password"
          required
          value={confirmPassword}
          onChange={(e) => setConfirmPassword(e.target.value)}
          error={confirmPassword && password !== confirmPassword ? "Passwords do not match" : undefined}
        />
        <Button type="submit" size="lg" loading={submitting}>
          Reset password
        </Button>
      </form>
    </>
  );
}

export default function ResetPasswordPage() {
  return (
    <div className="mx-auto flex max-w-md flex-col px-4 py-16 md:px-0">
      <Suspense fallback={null}>
        <ResetPasswordForm />
      </Suspense>
    </div>
  );
}

"use client";

import { Suspense, useEffect, useState } from "react";
import { useSearchParams, useRouter } from "next/navigation";
import { MailCheck, MailWarning } from "lucide-react";
import { Button, Loading, EmptyState } from "@saachi/ui";

function VerifyEmailContent() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const token = searchParams.get("token");
  const [status, setStatus] = useState<"checking" | "success" | "error">("checking");

  useEffect(() => {
    if (!token) {
      setStatus("error");
      return;
    }
    // NOTE: demo-only, no backend email-verification mutation wired up yet.
    const timer = setTimeout(() => setStatus("success"), 800);
    return () => clearTimeout(timer);
  }, [token]);

  if (status === "checking") {
    return <Loading label="Verifying your email..." />;
  }

  if (status === "error") {
    return (
      <EmptyState
        title="Verification link invalid"
        description="This email verification link is missing or has expired."
        action={<Button onClick={() => router.push("/login")}>Back to login</Button>}
      />
    );
  }

  return (
    <div className="flex flex-col items-center text-center">
      <div className="mb-4 flex h-14 w-14 items-center justify-center rounded-full bg-teal-light text-teal">
        <MailCheck size={26} />
      </div>
      <h1 className="mb-2 font-display text-[24px] font-bold text-ink">Email verified</h1>
      <p className="mb-8 max-w-sm text-[14px] text-ink-soft">
        Your email address has been confirmed. You can now access your account.
      </p>
      <Button onClick={() => router.push("/dashboard")}>Go to dashboard</Button>
    </div>
  );
}

export default function VerifyEmailPage() {
  return (
    <div className="mx-auto flex max-w-md flex-col px-4 py-20 md:px-0">
      <Suspense
        fallback={
          <div className="flex flex-col items-center text-center text-ink-soft">
            <MailWarning size={26} className="mb-3" />
            Loading...
          </div>
        }
      >
        <VerifyEmailContent />
      </Suspense>
    </div>
  );
}

import { Button, EmptyState, Input } from "@saachi/ui";

  const router = useRouter();

  const coupon = useCartStore((s) => s.coupon);
  const removeItem = useCartStore((s) => s.removeItem);
  const applyCoupon = useCartStore((s) => s.applyCoupon);
  const removeCoupon = useCartStore((s) => s.removeCoupon);
  const subtotal = useCartStore((s) => s.subtotal());
  const discount = useCartStore((s) => s.discount());
  const total = useCartStore((s) => s.total());

    const result = applyCoupon(couponCode);
    setCouponMessage(result.message);
  }

  const fmt = (n: number) => new Intl.NumberFormat("en-US", { style: "currency", currency }).format(n);

  return (
    <div>
      <h1 className="mb-6 font-display text-[26px] font-bold text-ink">Cart</h1>

          }
        />
      ) : (
        <div className="grid grid-cols-1 gap-8 md:grid-cols-[1fr_320px]">
          <div className="flex flex-col gap-3">
            {items.map((item) => (
              <div key={item.courseId} className="flex items-center gap-4 rounded-card border border-line bg-card p-4">

                  <X size={16} />

export default function CourseCommunityPage() {
  const params = useParams<{ courseId: string }>();
  const [questions, setQuestions] = useState(INITIAL_QUESTIONS);
  const [draft, setDraft] = useState("");
  const [upvotedIds, setUpvotedIds] = useState<Set<string>>(new Set());

  function handlePost() {
    if (!draft.trim()) return;
    const newQuestion: Question = {
      id: `q${Date.now()}`,
      authorName: "You",
      body: draft.trim(),
      createdAt: new Date().toISOString(),
      upvotes: 0
    };
    setQuestions((prev) => [newQuestion, ...prev]);
    setDraft("");
  }

  function toggleUpvote(id: string) {
    setUpvotedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) {
        next.delete(id);
      } else {
        next.add(id);
      }
      return next;
    });
    setQuestions((prev) =>
      prev.map((q) =>
        q.id === id ? { ...q, upvotes: q.upvotes + (upvotedIds.has(id) ? -1 : 1) } : q
      )
    );
  }

  return (
    <div className="mx-auto max-w-3xl px-4 py-10 md:px-8">
      <h1 className="mb-1 flex items-center gap-2 font-display text-[24px] font-bold text-ink">

      </p>

      <div className="mb-10 rounded-card border border-line bg-card p-5">
        <Textarea
          label="Ask a question"
          rows={3}
          value={draft}
          onChange={(e) => setDraft(e.target.value)}
          placeholder="What would you like to ask?"
        />
        <div className="mt-3 flex justify-end">
          <Button size="sm" disabled={!draft.trim()} onClick={handlePost}>
            <Send size={13} className="mr-1.5" />
            Post question
          </Button>
        </div>
