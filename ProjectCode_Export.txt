import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const PUBLIC_PATHS = ["/login"];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const adminSession = request.cookies.get("saachi_admin_session");

  const isPublic = PUBLIC_PATHS.some((p) => pathname.startsWith(p));

  if (!isPublic && !adminSession) {
    const loginUrl = new URL("/login", request.url);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"]
};

"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useState } from "react";
import Link from "next/link";
import { forgotPasswordSchema, type ForgotPasswordInput } from "@saachi/utils";
import { apiRequest } from "@saachi/api-client";
import { Button, Card } from "@saachi/ui";

export default function ForgotPasswordPage() {
  const [sent, setSent] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting }
  } = useForm<ForgotPasswordInput>({
    resolver: zodResolver(forgotPasswordSchema)
  });

  const onSubmit = async (data: ForgotPasswordInput) => {
    setServerError(null);
    try {
      await apiRequest("/auth/forgot-password", { method: "POST", body: data });
      setSent(true);
    } catch (err: any) {
      setServerError(err?.message ?? "Something went wrong. Please try again.");
    }
  };

  if (sent) {
    return (

        <div style={{ marginTop: "20px" }}>
          <Link href="/login">Back to login</Link>
        </div>
      </Card>
    );
  }

  return (
    <Card>
      <h1 style={{ fontSize: "22px", fontWeight: 700, marginBottom: "4px" }}>Reset your password</h1>
      <p style={{ color: "var(--text-secondary)", fontSize: "14px", marginBottom: "24px" }}>

        <div>
          <label style={{ display: "block", fontSize: "13px", fontWeight: 600, marginBottom: "6px" }}>
            Email
          </label>
          <input
            type="email"
            {...register("email")}
            style={{
              width: "100%",
              height: "44px",
              padding: "0 14px",
              borderRadius: "10px",
              border: "1px solid rgba(15,23,42,0.14)",
              fontSize: "15px"
            }}
            placeholder="you@example.com"
          />
          {errors.email && (
            <p style={{ color: "#EF4444", fontSize: "13px", marginTop: "4px" }}>{errors.email.message}</p>
          )}
        </div>

        {serverError && <p style={{ color: "#EF4444", fontSize: "13px" }}>{serverError}</p>}

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const LEARNER_PREFIX = "/dashboard";
const INSTRUCTOR_PREFIX = "/instructor";

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const sessionCookie = request.cookies.get("saachi_session");

  const isProtected =
    pathname.startsWith(LEARNER_PREFIX) || pathname.startsWith(INSTRUCTOR_PREFIX);

  if (isProtected && !sessionCookie) {
    const loginUrl = new URL("/login", request.url);
    loginUrl.searchParams.set("redirect", pathname);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/dashboard/:path*", "/instructor/:path*"]
};

export function useSession() {
  return useQuery({
    queryKey: authKeys.session,
    queryFn: () => apiRequest<Session>("/auth/session"),
    retry: false,
    staleTime: 60 * 1000
  });
}

export function useLogin() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (input: LoginInput) =>
      apiRequest<Session | { requires2fa: true }>("/auth/login", {
        method: "POST",
        body: input
      }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: authKeys.session });
    }
  });
}

export function useRegister() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (input: RegisterInput) =>
      apiRequest<Session>("/auth/register", { method: "POST", body: input }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: authKeys.session });
    }
  });
}

export function useLogout() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: () => apiRequest<void>("/auth/logout", { method: "POST" }),
    onSuccess: () => {
      queryClient.clear();
    }
  });
}

export type ApiError = {
  code: string;
  message: string;
  fieldErrors?: Record<string, string[]>;
};

export class ApiRequestError extends Error {
  code: string;
  fieldErrors?: Record<string, string[]>;
  status: number;

  constructor(status: number, error: ApiError) {
    super(error.message);
    this.name = "ApiRequestError";
    this.status = status;
    this.code = error.code;
    this.fieldErrors = error.fieldErrors;
  }
}

type RequestOptions = {
  method?: "GET" | "POST" | "PATCH" | "PUT" | "DELETE";
  body?: unknown;
  params?: Record<string, string | number | boolean | undefined>;
  signal?: AbortSignal;
};

function buildUrl(path: string, params?: RequestOptions["params"]): string {
  const base = process.env.NEXT_PUBLIC_API_URL ?? "";
  const url = new URL(path.replace(/^\//, ""), base.endsWith("/") ? base : `${base}/`);
  if (params) {
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) url.searchParams.set(key, String(value));
    });
  }
  return url.toString();
}

export async function apiRequest<TResponse>(
  path: string,
  options: RequestOptions = {}
): Promise<TResponse> {
  const { method = "GET", body, params, signal } = options;

  const response = await fetch(buildUrl(path, params), {
    method,
    credentials: "include",
    headers: {
      "Content-Type": "application/json"
    },
    body: body ? JSON.stringify(body) : undefined,
    signal
  });

  if (!response.ok) {
    let errorPayload: ApiError = { code: "unknown_error", message: "Something went wrong" };
    try {
      errorPayload = await response.json();
    } catch {
      // response had no JSON body
    }
    throw new ApiRequestError(response.status, errorPayload);
  }

  if (response.status === 204) return undefined as TResponse;
  return response.json() as Promise<TResponse>;
}

"use client";

import { Checkbox } from "./Checkbox";
import { Select } from "./Select";
import { cn } from "../cn";

export type FilterSidebarValue = {
  categories: string[];
  level?: string;
  language?: string;
};

export type FilterSidebarProps = {
  categoryOptions: { label: string; value: string }[];
  levelOptions: { label: string; value: string }[];
  languageOptions: { label: string; value: string }[];
  value: FilterSidebarValue;
  onChange: (next: FilterSidebarValue) => void;
  className?: string;
};

export function FilterSidebar({
  categoryOptions,
  levelOptions,
  languageOptions,
  value,
  onChange,
  className
}: FilterSidebarProps) {
  const toggleCategory = (cat: string) => {
    const isActive = value.categories.includes(cat);
    const nextCategories = isActive
      ? value.categories.filter((c) => c !== cat)
      : [...value.categories, cat];
    onChange({ ...value, categories: nextCategories });
  };

  return (
    <aside
      className={cn(
        "flex w-full flex-col gap-8 rounded-2xl border border-[var(--border-hairline,rgba(15,23,42,0.08))]",
        "bg-[var(--surface-card,#FFFFFF)] p-6",
        className
      )}
    >
      <div>
        <h4 className="mb-3 text-[13px] font-semibold uppercase tracking-[0.06em] text-[var(--text-muted,#94A3B8)]">
          Category
        </h4>
        <div className="flex flex-col gap-2.5">
          {categoryOptions.map((opt) => (
            <Checkbox
              key={opt.value}
              label={opt.label}
              checked={value.categories.includes(opt.value)}
              onChange={() => toggleCategory(opt.value)}
            />
          ))}
        </div>
      </div>

      <div>
        <Select
          label="Skill level"
          placeholder="All levels"
          options={levelOptions}
          value={value.level ?? ""}
          onChange={(e) => onChange({ ...value, level: e.target.value || undefined })}
        />
      </div>

      <div>
        <Select
          label="Language"
          placeholder="All languages"
          options={languageOptions}
          value={value.language ?? ""}
          onChange={(e) => onChange({ ...value, language: e.target.value || undefined })}
        />
      </div>

      <button
        type="button"
        onClick={() => onChange({ categories: [], level: undefined, language: undefined })}
        className="text-left text-[13px] font-medium text-[var(--blue-600,#2563EB)] hover:underline"
      >
        Clear all filters
      </button>
    </aside>
  );
}

"use client";

import { type ReactNode, useEffect } from "react";
import { X } from "lucide-react";
import { cn } from "../cn";

export type ModalProps = {
  open: boolean;
  onClose: () => void;
  title?: string;
  children: ReactNode;
  footer?: ReactNode;
  size?: "md" | "lg";
};

export function Modal({ open, onClose, title, children, footer, size = "md" }: ModalProps) {
  useEffect(() => {
    if (!open) return;
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", onKeyDown);
    return () => document.removeEventListener("keydown", onKeyDown);
  }, [open, onClose]);

  if (!open) return null;

  return (
    <div
      className="fixed inset-0 z-50 flex items-center justify-center p-4"
      role="dialog"
      aria-modal="true"
    >
      <div
        className="absolute inset-0 bg-[rgba(11,17,32,0.5)] backdrop-blur-[4px]"
        onClick={onClose}
      />
      <div
        className={cn(
          "relative w-full rounded-[24px] bg-[var(--surface-overlay,var(--surface-card,#FFFFFF))] p-6 shadow-[0_24px_64px_rgba(30,64,175,0.18)]",
          "animate-[modal-in_220ms_cubic-bezier(0.16,1,0.3,1)]",
          size === "lg" ? "max-w-[720px]" : "max-w-[520px]"
        )}
      >
        {title && (
          <div className="mb-4 flex items-center justify-between">
            <h2 className="text-[20px] font-semibold text-[var(--text-primary,#0F172A)]">{title}</h2>
            <button
              onClick={onClose}
              aria-label="Close"
              className="flex h-9 w-9 items-center justify-center rounded-full text-[var(--text-secondary,#475569)] transition-colors hover:bg-[var(--surface-elevated,#F1F5F9)]"
            >
              <X size={18} />
            </button>
          </div>
        )}
        <div>{children}</div>
        {footer && (
          <div className="mt-6 flex justify-end gap-3">{footer}</div>
        )}
      </div>
    </div>
  );
}

  USD: "en-US",
  PKR: "en-PK",
  GBP: "en-GB"
};

export function formatCurrency(amount: number, currency: string = "USD"): string {
  const locale = CURRENCY_LOCALE[currency] ?? "en-US";
  return new Intl.NumberFormat(locale, {
    style: "currency",
    currency,
    minimumFractionDigits: 2
  }).format(amount);
}

export function formatCompactNumber(value: number): string {
  return new Intl.NumberFormat("en-US", {
    notation: "compact",
    maximumFractionDigits: 1
  }).format(value);
}

export function formatDate(input: string | Date, locale: string = "en-US"): string {
  const date = typeof input === "string" ? new Date(input) : input;
  return new Intl.DateTimeFormat(locale, {
    year: "numeric",
    month: "short",
    day: "numeric"
  }).format(date);
}

export function formatRelativeTime(input: string | Date): string {
  const date = typeof input === "string" ? new Date(input) : input;
  const diffMs = date.getTime() - Date.now();
  const diffMinutes = Math.round(diffMs / 60000);

  const rtf = new Intl.RelativeTimeFormat("en", { numeric: "auto" });

  if (Math.abs(diffMinutes) < 60) return rtf.format(diffMinutes, "minute");
  const diffHours = Math.round(diffMinutes / 60);
  if (Math.abs(diffHours) < 24) return rtf.format(diffHours, "hour");
  const diffDays = Math.round(diffHours / 24);
  return rtf.format(diffDays, "day");
}

export function formatDuration(totalSeconds: number): string {
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  if (hours > 0) return `${hours}h ${minutes}m`;
  return `${minutes}m`;
}

import { z } from "zod";

export const loginSchema = z.object({
  email: z.string().email("Enter a valid email address"),
  password: z.string().min(8, "Password must be at least 8 characters")
});
export type LoginInput = z.infer<typeof loginSchema>;

export const registerSchema = z
  .object({
    fullName: z.string().min(2, "Full name is required"),
    email: z.string().email("Enter a valid email address"),
    password: z.string().min(8, "Password must be at least 8 characters"),
    confirmPassword: z.string(),
    role: z.enum(["learner", "instructor"]).default("learner")
  })
  .refine((data) => data.password === data.confirmPassword, {
    message: "Passwords do not match",
    path: ["confirmPassword"]
  });
export type RegisterInput = z.infer<typeof registerSchema>;

export const forgotPasswordSchema = z.object({
  email: z.string().email("Enter a valid email address")
});
export type ForgotPasswordInput = z.infer<typeof forgotPasswordSchema>;

export const resetPasswordSchema = z
  .object({
    token: z.string().min(1),
    password: z.string().min(8, "Password must be at least 8 characters"),
    confirmPassword: z.string()
  })
  .refine((data) => data.password === data.confirmPassword, {
    message: "Passwords do not match",
    path: ["confirmPassword"]
  });
export type ResetPasswordInput = z.infer<typeof resetPasswordSchema>;

export const twoFactorSchema = z.object({
  code: z.string().length(6, "Enter the 6-digit code")
});
export type TwoFactorInput = z.infer<typeof twoFactorSchema>;

import { z } from "zod";

export const courseBasicsSchema = z.object({
  title: z.string().min(5, "Title must be at least 5 characters").max(120),
  subtitle: z.string().max(180).optional(),
  categoryId: z.string().min(1, "Select a category"),
  tags: z.array(z.string()).max(10).default([]),
  language: z.string().min(1, "Select a language"),
  level: z.enum(["beginner", "intermediate", "advanced"])
});
export type CourseBasicsInput = z.infer<typeof courseBasicsSchema>;

export const coursePricingSchema = z.object({
  price: z.number().min(0, "Price cannot be negative"),
  currency: z.enum(["PKR", "USD", "GBP"]).default("USD"),
  subscriptionIncluded: z.boolean().default(false),
  couponEligible: z.boolean().default(true)
});
export type CoursePricingInput = z.infer<typeof coursePricingSchema>;

export const lessonSchema = z.object({
  title: z.string().min(3, "Lesson title is required"),
  type: z.enum(["video", "text", "quiz", "resource"]),
  content: z.string().optional(),
  durationSeconds: z.number().int().min(0).optional()
});
export type LessonInput = z.infer<typeof lessonSchema>;

export const moduleSchema = z.object({
  title: z.string().min(3, "Module title is required"),
  lessons: z.array(lessonSchema).default([])
});
export type ModuleInput = z.infer<typeof moduleSchema>;

export const curriculumSchema = z.object({
  modules: z.array(moduleSchema).min(1, "Add at least one module")
});
export type CurriculumInput = z.infer<typeof curriculumSchema>;
