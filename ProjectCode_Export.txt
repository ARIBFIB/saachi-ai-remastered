import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const PUBLIC_PATHS = ["/login"];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const adminSession = request.cookies.get("saachi_admin_session");

  const isPublic = PUBLIC_PATHS.some((p) => pathname.startsWith(p));

  if (!isPublic && !adminSession) {
    const loginUrl = new URL("/login", request.url);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"]
};

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useState } from "react";
import Link from "next/link";
import { forgotPasswordSchema, type ForgotPasswordInput } from "@saachi/utils";
import { apiRequest } from "@saachi/api-client";
import { Button, Card, Input } from "@saachi/ui";

export default function ForgotPasswordPage() {
  const [sent, setSent] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting }
  } = useForm<ForgotPasswordInput>({
    resolver: zodResolver(forgotPasswordSchema)
  });

  const onSubmit = async (data: ForgotPasswordInput) => {
    setServerError(null);
    try {
      await apiRequest("/auth/forgot-password", { method: "POST", body: data });
      setSent(true);
    } catch (err: any) {
      setServerError(err?.message ?? "Something went wrong. Please try again.");

  if (sent) {

          <Link href="/login" className="text-[14px] font-semibold text-ink hover:text-gold-dark">
            Back to login
          </Link>
        </div>
      </Card>
    );
  }

  return (
    <Card className="w-full max-w-[420px] p-8">
      <h1 className="font-display text-[22px] font-semibold text-ink">Reset your password</h1>
      <p className="mb-6 mt-1 text-[14px] text-ink-soft">
        Enter your email and we'll send you a reset link.
      </p>

          label="Email"
          type="email"
          placeholder="you@example.com"
          {...register("email")}
          error={errors.email?.message}

        {serverError && <p className="text-[13px] text-error">{serverError}</p>}

        <Button type="submit" size="lg" disabled={isSubmitting}>
          {isSubmitting ? "Sendingâ€¦" : "Send Reset Link"}

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const LEARNER_PREFIX = "/dashboard";
const INSTRUCTOR_PREFIX = "/instructor";

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const sessionCookie = request.cookies.get("saachi_session");

  const isProtected =
    pathname.startsWith(LEARNER_PREFIX) || pathname.startsWith(INSTRUCTOR_PREFIX);

  if (isProtected && !sessionCookie) {
    const loginUrl = new URL("/login", request.url);
    loginUrl.searchParams.set("redirect", pathname);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/dashboard/:path*", "/instructor/:path*"]
};

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { authKeys } from "../query-keys";
import type { Session } from "../types";
import type { LoginInput, RegisterInput } from "@saachi/utils";
import { DUMMY_SESSION } from "../dummy-data";

const SESSION_COOKIE = "saachi_session";
const SESSION_STORAGE_KEY = "saachi_demo_session";

function setSessionCookie() {
  if (typeof document === "undefined") return;
  document.cookie = `${SESSION_COOKIE}=demo_token; path=/; max-age=${60 * 60 * 24 * 7}`;
}

function clearSessionCookie() {
  if (typeof document === "undefined") return;
  document.cookie = `${SESSION_COOKIE}=; path=/; max-age=0`;
}

function readStoredSession(): Session | null {
  if (typeof window === "undefined") return null;
  const raw = window.localStorage.getItem(SESSION_STORAGE_KEY);
  if (!raw) return null;
  try {
    return JSON.parse(raw) as Session;
  } catch {
    return null;
  }
}

function storeSession(session: Session) {
  if (typeof window === "undefined") return;
  window.localStorage.setItem(SESSION_STORAGE_KEY, JSON.stringify(session));
}

function clearStoredSession() {
  if (typeof window === "undefined") return;
  window.localStorage.removeItem(SESSION_STORAGE_KEY);
}

function delay<T>(value: T, ms = 500): Promise<T> {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

// NOTE: Demo mode. There is no backend yet, so login/register accept ANY
// email/password and create a local mock session. Swap these mutationFns
// for real apiRequest(...) calls once the NestJS backend is available.

export function useSession() {
  return useQuery({
    queryKey: authKeys.session,
    queryFn: async () => readStoredSession(),
    staleTime: 60 * 1000
  });
}

export function useLogin() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (input: LoginInput) => {
      const session: Session = {
        ...DUMMY_SESSION,
        email: input.email || DUMMY_SESSION.email
      };
      return delay(session);
    },
    onSuccess: (session) => {
      storeSession(session);
      setSessionCookie();
      queryClient.setQueryData(authKeys.session, session);
    }
  });
}

export function useRegister() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (input: RegisterInput) => {
      const session: Session = {
        ...DUMMY_SESSION,
        fullName: input.fullName,
        email: input.email,
        role: input.role === "instructor" ? "instructor" : "learner"
      };
      return delay(session);
    },
    onSuccess: (session) => {
      storeSession(session);
      setSessionCookie();
      queryClient.setQueryData(authKeys.session, session);
    }
  });
}

export function useLogout() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async () => {
      clearStoredSession();
      clearSessionCookie();
      return delay(undefined, 200);
    },
    onSuccess: () => {
      queryClient.clear();
    }
  });
}

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { courseKeys } from "../query-keys";
import type { Course, CourseDetail, Paginated } from "../types";
import { DUMMY_COURSES, type DummyCourse } from "../dummy-data";

export type CourseListFilters = {
  category?: string;
  search?: string;
  priceMin?: number;
  priceMax?: number;
  level?: string;
  language?: string;
  sort?: string;
  page?: number;
};

function delay<T>(value: T, ms = 400): Promise<T> {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

function toSummary(course: DummyCourse): Course {
  return {
    id: course.id,
    slug: course.slug,
    title: course.title,
    subtitle: course.subtitle,
    thumbnailUrl: course.thumbnailUrl,
    price: course.price,
    currency: course.currency,
    rating: course.rating,
    ratingCount: course.ratingCount,
    instructor: course.instructor,
    status: course.status
  };
}

export function useCourseList(filters: CourseListFilters = {}) {
  return useQuery({
    queryKey: courseKeys.list(filters),
    queryFn: async () => {
      let items = [...DUMMY_COURSES];

      if (filters.search) {
        const q = filters.search.toLowerCase();
        items = items.filter(
          (c) => c.title.toLowerCase().includes(q) || c.instructor.fullName.toLowerCase().includes(q)
        );
      }
      if (filters.category) {
        items = items.filter((c) => c.categoryId === filters.category);
      }
      if (filters.level) {
        items = items.filter((c) => c.level === filters.level);
      }
      if (filters.language) {
        items = items.filter((c) => c.language === filters.language);
      }
      if (typeof filters.priceMin === "number") {
        items = items.filter((c) => c.price >= (filters.priceMin as number));
      }
      if (typeof filters.priceMax === "number") {
        items = items.filter((c) => c.price <= (filters.priceMax as number));
      }

      switch (filters.sort) {
        case "newest":
          items = [...items].reverse();
          break;
        case "price_asc":
          items = [...items].sort((a, b) => a.price - b.price);
          break;
        case "price_desc":
          items = [...items].sort((a, b) => b.price - a.price);
          break;
        case "rating":
          items = [...items].sort((a, b) => b.rating - a.rating);
          break;
        case "popular":
        default:
          items = [...items].sort((a, b) => b.ratingCount - a.ratingCount);
      }

      const page = filters.page ?? 1;
      const pageSize = 12;
      const start = (page - 1) * pageSize;
      const pageItems = items.slice(start, start + pageSize).map(toSummary);

      return delay<Paginated<Course>>({
        items: pageItems,
        total: items.length,
        page,
        pageSize
      });
    }
  });
}

export function useCourseDetail(idOrSlug: string) {
  return useQuery({
    queryKey: courseKeys.detail(idOrSlug),
    queryFn: async () => {
      const found = DUMMY_COURSES.find((c) => c.slug === idOrSlug || c.id === idOrSlug);
      if (!found) throw new Error("Course not found");
      const { categoryId, level, language, ...detail } = found;
      return delay<CourseDetail>(detail);
    },
    enabled: Boolean(idOrSlug)
  });
}

export function useCreateCourse() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (input: Partial<Course>) => {
      const created: Course = {
        id: `crs_${Math.random().toString(36).slice(2, 8)}`,
        slug: (input.slug as string) ?? "new-course",
        title: input.title ?? "Untitled course",
        price: input.price ?? 0,
        currency: input.currency ?? "USD",
        rating: 0,
        ratingCount: 0,
        instructor: input.instructor ?? { id: "ins_demo", fullName: "Demo Instructor" },
        status: "draft"
      };
      return delay(created);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: courseKeys.all });
    }
  });
}

export function useUpdateCourse(id: string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async (input: Partial<Course>) => {
      const existing = DUMMY_COURSES.find((c) => c.id === id);
      const base: Course = existing
        ? toSummary(existing)
        : {
            id,
            slug: id,
            title: "Untitled",
            price: 0,
            currency: "USD",
            rating: 0,
            ratingCount: 0,
            instructor: { id: "ins_demo", fullName: "Demo Instructor" },
            status: "draft"
          };
      const updated: Course = { ...base, ...input };
      return delay(updated);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: courseKeys.detail(id) });
      queryClient.invalidateQueries({ queryKey: courseKeys.all });
    }
  });
}

export type ApiError = {
  code: string;
  message: string;
  fieldErrors?: Record<string, string[]>;
};

export class ApiRequestError extends Error {
  code: string;
  fieldErrors?: Record<string, string[]>;
  status: number;

  constructor(status: number, error: ApiError) {
    super(error.message);
    this.name = "ApiRequestError";
    this.status = status;
    this.code = error.code;
    this.fieldErrors = error.fieldErrors;
  }
}

type RequestOptions = {
  method?: "GET" | "POST" | "PATCH" | "PUT" | "DELETE";
  body?: unknown;
  params?: Record<string, string | number | boolean | undefined>;
  signal?: AbortSignal;
};

function buildUrl(path: string, params?: RequestOptions["params"]): string {
  const base = process.env.NEXT_PUBLIC_API_URL ?? "";
  const url = new URL(path.replace(/^\//, ""), base.endsWith("/") ? base : `${base}/`);
  if (params) {
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) url.searchParams.set(key, String(value));
    });
  }
  return url.toString();
}

export async function apiRequest<TResponse>(
  path: string,
  options: RequestOptions = {}
): Promise<TResponse> {
  const { method = "GET", body, params, signal } = options;

  const response = await fetch(buildUrl(path, params), {
    method,
    credentials: "include",
    headers: {
      "Content-Type": "application/json"
    },
    body: body ? JSON.stringify(body) : undefined,
    signal
  });

  if (!response.ok) {
    let errorPayload: ApiError = { code: "unknown_error", message: "Something went wrong" };
    try {
      errorPayload = await response.json();
    } catch {
      // response had no JSON body
    }
    throw new ApiRequestError(response.status, errorPayload);
  }

  if (response.status === 204) return undefined as TResponse;
  return response.json() as Promise<TResponse>;
}

  defaultOpenId?: string;
  className?: string;

                {item.meta}
              </div>

            </button>

export type FilterSidebarValue = {

export type FilterSidebarProps = {
  categoryOptions: { label: string; value: string }[];
  levelOptions: { label: string; value: string }[];
  languageOptions: { label: string; value: string }[];
  value: FilterSidebarValue;
  onChange: (next: FilterSidebarValue) => void;
  className?: string;
};

export function FilterSidebar({
  categoryOptions,
  levelOptions,
  languageOptions,
  value,
  onChange,
  className
}: FilterSidebarProps) {
  const toggleCategory = (cat: string) => {
    const isActive = value.categories.includes(cat);
    const nextCategories = isActive
      ? value.categories.filter((c) => c !== cat)
      : [...value.categories, cat];
    onChange({ ...value, categories: nextCategories });
  };

  return (
    <aside className={cn("flex w-full flex-col gap-8 rounded-card border border-line bg-card p-6", className)}>
      <div>
        <h4 className="mb-3 text-[13px] font-semibold uppercase tracking-[0.06em] text-ink-soft/70">
          Category
        </h4>
        <div className="flex flex-col gap-2.5">
          {categoryOptions.map((opt) => (
            <Checkbox
              key={opt.value}
              label={opt.label}
              checked={value.categories.includes(opt.value)}
              onChange={() => toggleCategory(opt.value)}
            />
          ))}
        </div>
      </div>

      <Select
        label="Skill level"
        placeholder="All levels"
        options={levelOptions}
        value={value.level ?? ""}
        onChange={(e) => onChange({ ...value, level: e.target.value || undefined })}
      />

      <Select
        label="Language"
        placeholder="All languages"
        options={languageOptions}
        value={value.language ?? ""}
        onChange={(e) => onChange({ ...value, language: e.target.value || undefined })}
      />

      <button
        type="button"
        onClick={() => onChange({ categories: [], level: undefined, language: undefined })}
        className="text-left text-[13px] font-medium text-gold-dark hover:underline"
      >
        Clear all filters
      </button>
    </aside>
  );
}

"use client";

import { type ReactNode, useEffect } from "react";
import { X } from "lucide-react";
import { cn } from "../cn";

export type ModalProps = {
  open: boolean;
  onClose: () => void;
  title?: string;
  children: ReactNode;
  footer?: ReactNode;
  size?: "md" | "lg";
};

export function Modal({ open, onClose, title, children, footer, size = "md" }: ModalProps) {
  useEffect(() => {
    if (!open) return;
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", onKeyDown);
    return () => document.removeEventListener("keydown", onKeyDown);
  }, [open, onClose]);

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4" role="dialog" aria-modal="true">
      <div className="absolute inset-0 bg-ink/40" onClick={onClose} />
      <div
        className={cn(
          "relative w-full rounded-card border border-line bg-card p-6",
          size === "lg" ? "max-w-[720px]" : "max-w-[520px]"
        )}
      >
        {title && (
          <div className="mb-4 flex items-center justify-between">
            <h2 className="font-display text-[20px] font-semibold text-ink">{title}</h2>
            <button
              onClick={onClose}
              aria-label="Close"
              className="flex h-9 w-9 items-center justify-center rounded-full text-ink-soft transition-colors hover:bg-paper-dim"
            >
              <X size={18} />
            </button>
          </div>
        )}
        <div>{children}</div>
        {footer && <div className="mt-6 flex justify-end gap-3">{footer}</div>}
      </div>
    </div>
  );
}

export function formatDate(input: string | Date, locale: string = "en-US"): string {
  const date = typeof input === "string" ? new Date(input) : input;
  return new Intl.DateTimeFormat(locale, {
    year: "numeric",
    month: "short",
    day: "numeric"
  }).format(date);
}

export function formatRelativeTime(input: string | Date): string {
  const date = typeof input === "string" ? new Date(input) : input;
  const diffMs = date.getTime() - Date.now();
  const diffMinutes = Math.round(diffMs / 60000);

  const rtf = new Intl.RelativeTimeFormat("en", { numeric: "auto" });

  if (Math.abs(diffMinutes) < 60) return rtf.format(diffMinutes, "minute");
  const diffHours = Math.round(diffMinutes / 60);
  if (Math.abs(diffHours) < 24) return rtf.format(diffHours, "hour");
  const diffDays = Math.round(diffHours / 24);
  return rtf.format(diffDays, "day");
}

export function formatDuration(totalSeconds: number): string {
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  if (hours > 0) return `${hours}h ${minutes}m`;
  return `${minutes}m`;
}
