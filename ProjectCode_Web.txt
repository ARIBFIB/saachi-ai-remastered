"use client";

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { useState } from "react";
import Link from "next/link";
import { forgotPasswordSchema, type ForgotPasswordInput } from "@saachi/utils";
import { apiRequest } from "@saachi/api-client";
import { Button, Card, Input } from "@saachi/ui";

export default function ForgotPasswordPage() {
  const [sent, setSent] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors, isSubmitting }
  } = useForm<ForgotPasswordInput>({
    resolver: zodResolver(forgotPasswordSchema)
  });

  const onSubmit = async (data: ForgotPasswordInput) => {
    setServerError(null);
    try {
      await apiRequest("/auth/forgot-password", { method: "POST", body: data });
      setSent(true);
    } catch (err: any) {
      setServerError(err?.message ?? "Something went wrong. Please try again.");
    }
  };

  if (sent) {
    return (
      <Card className="w-full max-w-[420px] p-8">
        <h1 className="font-display text-[22px] font-semibold text-ink">Check your email</h1>

          <Link href="/login" className="text-[14px] font-semibold text-ink hover:text-gold-dark">
            Back to login
          </Link>
        </div>
      </Card>
    );
  }

  return (
    <Card className="w-full max-w-[420px] p-8">
      <h1 className="font-display text-[22px] font-semibold text-ink">Reset your password</h1>

          label="Email"
          type="email"
          placeholder="you@example.com"

        <Button type="submit" size="lg" disabled={isSubmitting}>

"use client";

import { useState } from "react";
import { useCorporateStore } from "@saachi/store";
import { DUMMY_COURSES } from "@saachi/api-client";
import { Button, Card, Select, Badge, EmptyState } from "@saachi/ui";
import { Users } from "lucide-react";

export default function CourseAssignmentPage() {
  const { team, assignCourse } = useCorporateStore();
  const [selectedMemberId, setSelectedMemberId] = useState(team[0]?.id ?? "");
  const [selectedCourseId, setSelectedCourseId] = useState(DUMMY_COURSES[0]?.id ?? "");

  const memberOptions = team.map((m) => ({ label: m.name, value: m.id }));
  const courseOptions = DUMMY_COURSES.map((c) => ({ label: c.title, value: c.id }));

  const handleAssign = () => {
    if (!selectedMemberId || !selectedCourseId) return;
    assignCourse(selectedMemberId, selectedCourseId);
  };

  return (
    <main className="mx-auto max-w-[900px] px-6 py-10 md:px-10">
      <h1 className="mb-8 font-display text-[26px] font-semibold text-ink">Course Assignment</h1>

      {team.length === 0 ? (
        <EmptyState icon={<Users size={22} />} title="No team members yet" description="Invite team members first from Team Roster." />
      ) : (
        <>
          <Card className="mb-8 p-6">
            <h2 className="mb-4 text-[15px] font-semibold text-ink">Assign a course</h2>
            <div className="flex flex-col gap-4 sm:flex-row sm:items-end">
              <Select label="Team member" options={memberOptions} value={selectedMemberId} onChange={(e) => setSelectedMemberId(e.target.value)} className="flex-1" />
              <Select label="Course" options={courseOptions} value={selectedCourseId} onChange={(e) => setSelectedCourseId(e.target.value)} className="flex-1" />
              <Button onClick={handleAssign}>Assign</Button>
            </div>
          </Card>

          <div className="flex flex-col gap-3">
            {team.map((member) => (
              <Card key={member.id} className="p-4">
                <p className="mb-2 text-[14px] font-semibold text-ink">{member.name}</p>
                {member.assignedCourseIds.length === 0 ? (
                  <p className="text-[12px] text-ink-soft">No courses assigned yet.</p>
                ) : (
                  <div className="flex flex-wrap gap-2">
                    {member.assignedCourseIds.map((courseId) => {
                      const course = DUMMY_COURSES.find((c) => c.id === courseId);
                      const completed = member.completedCourseIds.includes(courseId);
                      return (
                        <Badge key={courseId} variant={completed ? "success" : "default"}>
                          {course?.title ?? courseId} {completed ? "- done" : ""}
                        </Badge>
                      );
                    })}
                  </div>
                )}
              </Card>
            ))}
          </div>
        </>
      )}
    </main>
  );
}

export default function CorporateDashboardPage() {
  const { account, team } = useCorporateStore();

  const totalAssignments = team.reduce((sum, m) => sum + m.assignedCourseIds.length, 0);
  const totalCompletions = team.reduce((sum, m) => sum + m.completedCourseIds.length, 0);
  const completionRate = totalAssignments > 0 ? Math.round((totalCompletions / totalAssignments) * 100) : 0;
  const activeMembers = team.filter((m) => m.status === "active").length;

  const stats = [
    { label: "Team members", value: `${activeMembers}/${team.length}`, icon: Users },
    { label: "Seats available", value: `${account.seatsPurchased - account.seatsUsed}`, icon: CreditCard },
    { label: "Course assignments", value: String(totalAssignments), icon: BookOpen },
    { label: "Completion rate", value: `${completionRate}%`, icon: TrendingUp }

  return (
    <main className="mx-auto max-w-[1000px] px-6 py-10 md:px-10">
      <div className="mb-8">
        <h1 className="font-display text-[26px] font-semibold text-ink">{account.companyName} Dashboard</h1>

"use client";

import { useState } from "react";
import { UserPlus, Trash2 } from "lucide-react";
import { useCorporateStore } from "@saachi/store";
import { Button, Card, Input, Badge } from "@saachi/ui";

export default function TeamRosterPage() {
  const { team, account, inviteMember, removeMember } = useCorporateStore();
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");

  const seatsRemaining = account.seatsPurchased - account.seatsUsed;

  const handleInvite = (e: React.FormEvent) => {
    e.preventDefault();
    if (!name.trim() || !email.trim() || seatsRemaining <= 0) return;
    inviteMember(name.trim(), email.trim());
    setName("");
    setEmail("");
  };

  return (
    <main className="mx-auto max-w-[900px] px-6 py-10 md:px-10">
      <div className="mb-8">
        <h1 className="font-display text-[26px] font-semibold text-ink">Team Roster</h1>
        <p className="mt-1 text-[14px] text-ink-soft">
          {seatsRemaining} of {account.seatsPurchased} seats available
        </p>
      </div>

      <Card className="mb-8 p-6">
        <h2 className="mb-4 text-[15px] font-semibold text-ink">Invite a team member</h2>
        <form onSubmit={handleInvite} className="flex flex-col gap-4 sm:flex-row sm:items-end">
          <Input label="Full name" placeholder="Team member's name" value={name} onChange={(e) => setName(e.target.value)} className="flex-1" />
          <Input label="Work email" type="email" placeholder="name@company.com" value={email} onChange={(e) => setEmail(e.target.value)} className="flex-1" />

      <div className="flex flex-col gap-3">

import { useMemo, useState } from "react";
import { useSearchParams, useRouter } from "next/navigation";
import { Search } from "lucide-react";
import Link from "next/link";
import { DUMMY_COURSES } from "@saachi/api-client";
import { CourseCard, Input, EmptyState, Card } from "@saachi/ui";
import { Navbar } from "../_components/Navbar";
import { Footer } from "../_components/Footer";

function slugifyName(name: string) {
  return name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "");
}

export default function SearchPage() {
  const params = useSearchParams();
  const router = useRouter();
  const initialQuery = params.get("q") ?? "";
  const [query, setQuery] = useState(initialQuery);

  const results = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return { courses: [], instructors: [] };

    const courses = DUMMY_COURSES.filter((c) => c.title.toLowerCase().includes(q));

    const instructorMap = new Map<string, { name: string; slug: string; avatarUrl?: string; courseCount: number }>();
    DUMMY_COURSES.forEach((c) => {
      if (c.instructor.fullName.toLowerCase().includes(q)) {
        const slug = slugifyName(c.instructor.fullName);
        const existing = instructorMap.get(slug);
        if (existing) {
          existing.courseCount += 1;
        } else {
          instructorMap.set(slug, {
            name: c.instructor.fullName,
            slug,
            avatarUrl: c.instructor.avatarUrl,
            courseCount: 1
          });
        }
      }
    });

    return { courses, instructors: Array.from(instructorMap.values()) };
  }, [query]);

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    router.push(`/search?q=${encodeURIComponent(query)}`);
  };

  const hasResults = results.courses.length > 0 || results.instructors.length > 0;

  return (
    <>
      <Navbar />
      <main className="mx-auto max-w-[1000px] px-6 py-16">
        <h1 className="mb-6 font-display text-[28px] font-semibold text-ink">Search</h1>

        <form onSubmit={handleSubmit} className="relative mb-10 max-w-[520px]">
          <Search size={18} className="pointer-events-none absolute left-3.5 top-1/2 -translate-y-1/2 text-ink-soft/60" />
          <Input
            placeholder="Search courses and instructors"
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            className="pl-10"
          />
        </form>

        {!query.trim() ? (
          <EmptyState title="Start typing to search" description="Search across all courses and instructors." />
        ) : !hasResults ? (
          <EmptyState title="No results found" description={`Nothing matched "${query}". Try a different term.`} />
        ) : (
          <div className="flex flex-col gap-14">
            {results.instructors.length > 0 && (
              <div>
                <h2 className="mb-4 text-[16px] font-semibold text-ink">Instructors</h2>
                <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
                  {results.instructors.map((instructor) => (
                    <Link key={instructor.slug} href={`/instructors/${instructor.slug}`}>
                      <Card className="flex items-center gap-4 p-4">
                        <div className="flex h-12 w-12 shrink-0 items-center justify-center overflow-hidden rounded-full bg-teal-light text-[15px] font-semibold text-teal">
                          {instructor.avatarUrl ? (
                            // eslint-disable-next-line @next/next/no-img-element
                            <img src={instructor.avatarUrl} alt={instructor.name} className="h-full w-full object-cover" />
                          ) : (
                            instructor.name.charAt(0)
                          )}
                        </div>
                        <div>
                          <p className="text-[14px] font-semibold text-ink">{instructor.name}</p>
                          <p className="text-[12px] text-ink-soft">{instructor.courseCount} courses</p>
                        </div>

"use client";

import { useEffect } from "react";
import { useNotificationStore } from "@saachi/store";

const DEMO_MESSAGES = [
  { title: "New course recommendation", body: "Based on your activity, you might like AI Automation & Workflow Tools.", type: "system" as const },
  { title: "Instructor replied", body: "Your question in the course Q&A received a new reply.", type: "community" as const },
  { title: "Payment reminder", body: "A saved cart item has a limited-time coupon expiring soon.", type: "payment" as const }
];

export function useRealtimeNotifications(enabled: boolean) {
  useEffect(() => {
    if (!enabled) return;

    const interval = setInterval(() => {
      const msg = DEMO_MESSAGES[Math.floor(Math.random() * DEMO_MESSAGES.length)];
      const store = useNotificationStore.getState();
      const alreadyExists = store.items.some((n) => n.title === msg.title && !n.read);
      if (alreadyExists) return;

      useNotificationStore.setState({
        items: [
          {
            id: `ntf_rt_${Date.now()}`,
            title: msg.title,
            body: msg.body,
            type: msg.type,
            createdAt: new Date().toISOString(),
            read: false
          },
          ...store.items
        ]
      });
    }, 45000);

    return () => clearInterval(interval);
  }, [enabled]);
}

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const LEARNER_PREFIX = "/dashboard";
const INSTRUCTOR_PREFIX = "/instructor";

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const sessionCookie = request.cookies.get("saachi_session");

  const isProtected =
    pathname.startsWith(LEARNER_PREFIX) || pathname.startsWith(INSTRUCTOR_PREFIX);

  if (isProtected && !sessionCookie) {
    const loginUrl = new URL("/login", request.url);
    loginUrl.searchParams.set("redirect", pathname);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/dashboard/:path*", "/instructor/:path*"]
};
