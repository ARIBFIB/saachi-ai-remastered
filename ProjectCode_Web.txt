function Checkbox({
  label,
  checked,
  onChange,
}: {
  label: string;
  checked: boolean;
  onChange: () => void;
}) {
  return (
    <label className="flex cursor-pointer items-center gap-2">
      <button
        type="button"
        onClick={onChange}
        className={`flex h-4 w-4 items-center justify-center rounded-sm border transition ${
          checked
            ? "border-[#3525CD] bg-[#3525CD]"
            : "border-[#C7C4D8] bg-white"
        }`}
      >
        {checked && <Check className="h-3 w-3 text-white" strokeWidth={3} />}
      </button>
      <span className="text-base text-slate-600">{label}</span>
    </label>
  );
}

export function FilterSidebar() {
  const [selectedCategories, setSelectedCategories] = useState<string[]>([
    "Machine Learning",
  ]);
  const [selectedLevels, setSelectedLevels] = useState<string[]>([
    "Intermediate",
  ]);

  function toggle(
    list: string[],
    setList: (v: string[]) => void,
    value: string
  ) {
    setList(
      list.includes(value) ? list.filter((v) => v !== value) : [...list, value]
    );
  }

  return (
    <aside className="w-56 rounded-xl border border-slate-300/30 bg-white p-6 shadow-[0px_4px_20px_0px_rgba(15,23,42,0.05)]">
      <div className="flex flex-col gap-4">
        <h3 className="text-sm font-semibold tracking-wide text-zinc-900">
          Category
        </h3>
        <div className="flex flex-col gap-2">
          {categories.map((c) => (
            <Checkbox
              key={c}
              label={c}
              checked={selectedCategories.includes(c)}
              onChange={() => toggle(selectedCategories, setSelectedCategories, c)}
            />
          ))}
        </div>
      </div>

      <hr className="my-6 border-slate-300/50" />

              label={l}
              checked={selectedLevels.includes(l)}
              onChange={() => toggle(selectedLevels, setSelectedLevels, l)}
            />
          ))}
        </div>
      </div>

      <hr className="my-6 border-slate-300/50" />

      <div className="flex flex-col gap-4">
        <h3 className="text-sm font-semibold tracking-wide text-zinc-900">

import { ChevronLeft, ChevronRight } from "lucide-react";

export function Pagination({ totalPages = 12 }: { totalPages?: number }) {
  const [page, setPage] = useState(1);

  const pages = [1, 2, 3];

  return (
    <div className="flex items-center justify-center gap-2">
      <button
        aria-label="Previous page"
        onClick={() => setPage((p) => Math.max(1, p - 1))}
        className="flex h-10 w-8 items-center justify-center border border-slate-300 text-slate-600 hover:bg-gray-100"
      >
        <ChevronLeft className="h-3 w-3" />

        <button
          key={p}
          onClick={() => setPage(p)}
          className={`flex h-10 w-10 items-center justify-center text-sm font-semibold tracking-wide ${
            page === p
              ? "bg-[#3525CD] font-bold text-white shadow-[0px_1px_2px_0px_rgba(0,0,0,0.05)]"
              : "border border-slate-300 text-slate-600 hover:bg-gray-100"
          }`}
        >
          {p}
        </button>
      ))}

      <span className="px-1 text-base text-gray-500">...</span>

export type BillingCycle = "monthly" | "annually";

interface BillingToggleProps {
  onChange?: (cycle: BillingCycle) => void;
}

export function BillingToggle({ onChange }: BillingToggleProps) {
  const [cycle, setCycle] = useState<BillingCycle>("monthly");

  function toggle() {
    const next = cycle === "monthly" ? "annually" : "monthly";
    setCycle(next);
    onChange?.(next);
  }

  const isAnnual = cycle === "annually";

  return (
    <div className="flex items-center justify-center gap-4 pt-4">
      <span className="text-sm font-semibold tracking-wide text-[var(--token-color-ink)]">Monthly</span>

      <button
        type="button"
        role="switch"

// ============================================================================
// generate-css-vars.mjs
// Syncs lib/design-system/tokens.ts -> app/globals.css (Tailwind @theme block)
// Runs automatically before dev/build via npm scripts. Do not edit the
// generated block in globals.css by hand -- edit tokens.ts instead.
// ============================================================================
import { readFileSync, writeFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import path from "node:path";

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const tokensPath = path.join(__dirname, "tokens.ts");
const cssPath = path.join(__dirname, "..", "..", "app", "globals.css");

const START_MARKER = "/* === AUTO-GENERATED TOKENS START (do not edit) === */";
const END_MARKER = "/* === AUTO-GENERATED TOKENS END === */";

async function loadTokens() {
  // Dynamic import works because tokens.ts is plain TS object literal,
  // executed here via a lightweight transpile-free eval of the exported object.
  const raw = readFileSync(tokensPath, "utf-8");
  const match = raw.match(/export const tokens = ({[\s\S]*?}) as const;/);
  if (!match) throw new Error("Could not parse tokens.ts");
  // eslint-disable-next-line no-eval
  const tokens = eval(`(${match[1]})`);
  return tokens;
}

function flatten(obj, prefix = "") {
  const out = {};
  for (const [key, value] of Object.entries(obj)) {
    const varName = prefix ? `${prefix}-${key}` : key;
    if (typeof value === "object" && value !== null) {
      Object.assign(out, flatten(value, varName));
    } else {
      out[varName] = value;
    }
  }
  return out;
}

const tokens = await loadTokens();
const flat = flatten(tokens);

const cssVars = Object.entries(flat)
  .map(([k, v]) => `  --token-${k}: ${v};`)
  .join("\n");

const block = `${START_MARKER}\n@theme {\n${cssVars}\n}\n${END_MARKER}`;

let css = readFileSync(cssPath, "utf-8");

if (css.includes(START_MARKER) && css.includes(END_MARKER)) {
  const regex = new RegExp(`${START_MARKER}[\\s\\S]*?${END_MARKER}`);
  css = css.replace(regex, block);
} else {
  css = `${block}\n\n${css}`;
}

writeFileSync(cssPath, css, "utf-8");
console.log(`[tokens:sync] ${Object.keys(flat).length} tokens written to globals.css`);

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const LEARNER_PREFIX = "/dashboard";
const INSTRUCTOR_PREFIX = "/instructor";

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const sessionCookie = request.cookies.get("saachi_session");

  const isProtected =
    pathname.startsWith(LEARNER_PREFIX) || pathname.startsWith(INSTRUCTOR_PREFIX);

  if (isProtected && !sessionCookie) {
    const loginUrl = new URL("/login", request.url);
    loginUrl.searchParams.set("redirect", pathname);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/dashboard/:path*", "/instructor/:path*"]
};
