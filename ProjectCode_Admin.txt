"use client";

import { useState } from "react";
import { Tags, Plus, Pencil, Trash2, X } from "lucide-react";
import {
  useAdminCategoryList,
  useCreateCategory,
  useUpdateCategory,
  useDeleteCategory,
  type AdminCategory
} from "@saachi/api-client";
import { Button, Card, Input, Modal, ShimmerTableRow, Badge , HelpTooltip } from "@saachi/ui";

export default function CategoryManagementPage() {
  const { data: categories, isLoading } = useAdminCategoryList();
  const createCategory = useCreateCategory();
  const updateCategory = useUpdateCategory();
  const deleteCategory = useDeleteCategory();

  const [editing, setEditing] = useState<AdminCategory | null>(null);
  const [showCreate, setShowCreate] = useState(false);
  const [name, setName] = useState("");
  const [tagInput, setTagInput] = useState("");
  const [tags, setTags] = useState<string[]>([]);
  const [deleteTarget, setDeleteTarget] = useState<AdminCategory | null>(null);

  const openCreate = () => {
    setName("");
    setTags([]);
    setTagInput("");
    setShowCreate(true);
  };

  const openEdit = (category: AdminCategory) => {
    setEditing(category);
    setName(category.name);
    setTags(category.tags);
    setTagInput("");
  };

  const addTag = () => {
    const value = tagInput.trim().toLowerCase();
    if (!value || tags.includes(value)) return;
    setTags((prev) => [...prev, value]);
    setTagInput("");
  };

  const handleSave = () => {
    if (!name.trim()) return;
    if (editing) {
      updateCategory.mutate({ id: editing.id, name: name.trim(), tags });
      setEditing(null);
    } else {
      createCategory.mutate({ name: name.trim(), tags });
      setShowCreate(false);
    }
  };

  const handleDelete = () => {
    if (!deleteTarget) return;
    deleteCategory.mutate(deleteTarget.id);
    setDeleteTarget(null);
  };

  const modalOpen = showCreate || Boolean(editing);

  return (
    <main className="mx-auto max-w-[900px] px-4 py-8 md:px-8 md:py-10">
      <div className="mb-6 flex items-center justify-between gap-3">
        <div className="flex items-center gap-3">
          <div className="flex h-10 w-10 items-center justify-center rounded-control border border-line text-ink">
            <Tags size={17} strokeWidth={1.75} />
          </div>
          <div>
            <div className="flex items-center gap-1.5"><h1 className="text-[22px] font-semibold text-ink">Category & Tag Management</h1><HelpTooltip text="The taxonomy used across the marketplace and course builder. Deleting a category will require recategorizing any courses assigned to it." /></div>
            <p className="mt-0.5 text-[13px] text-ink-soft">Taxonomy used across the marketplace and course builder.</p>
          </div>
        </div>
        <Button size="sm" onClick={openCreate}>
          <Plus size={14} strokeWidth={1.75} /> New category
        </Button>
      </div>

      <div className="overflow-hidden rounded-card border border-line bg-card">
        {isLoading ? (
          Array.from({ length: 5 }).map((_, i) => <ShimmerTableRow key={i} columns={3} />)
        ) : (
          categories?.map((category) => (
            <div key={category.id} className="flex flex-col gap-3 border-b border-line px-5 py-4 last:border-0 sm:flex-row sm:items-center sm:justify-between">
              <div>
                <div className="flex items-center gap-2.5">
                  <p className="text-[14px] font-semibold text-ink">{category.name}</p>
                  <span className="text-[12px] text-ink-soft">/{category.slug}</span>
                </div>
                <div className="mt-2 flex flex-wrap gap-1.5">
                  {category.tags.map((tag) => (
                    <Badge key={tag}>{tag}</Badge>
                  ))}
                </div>
              </div>
              <div className="flex shrink-0 items-center gap-4">
                <span className="text-[12px] text-ink-soft">{category.courseCount} courses</span>
                <div className="flex gap-1.5">
                  <button
                    type="button"
                    onClick={() => openEdit(category)}
                    className="flex h-8 w-8 items-center justify-center rounded-control text-ink-soft hover:bg-paper-dim hover:text-ink"
                  >
                    <Pencil size={14} strokeWidth={1.75} />
                  </button>
                  <button
                    type="button"
                    onClick={() => setDeleteTarget(category)}
                    className="flex h-8 w-8 items-center justify-center rounded-control text-ink-soft hover:bg-paper-dim hover:text-error"
                  >
                    <Trash2 size={14} strokeWidth={1.75} />
                  </button>
                </div>
              </div>
            </div>
          ))
        )}
      </div>

      <Modal
        open={modalOpen}
        onClose={() => {
          setShowCreate(false);
          setEditing(null);
        }}
        title={editing ? "Edit category" : "New category"}
        footer={
          <>
            <Button
              variant="ghost"
              onClick={() => {
                setShowCreate(false);
                setEditing(null);
              }}
            >
              Cancel
            </Button>
            <Button onClick={handleSave} disabled={!name.trim() || createCategory.isPending || updateCategory.isPending}>
              {editing ? "Save changes" : "Create category"}
            </Button>
          </>
        }
      >
        <div className="flex flex-col gap-4">
          <Input label="Category name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. AI for Careers & Job Search" />
          <div>
            <label className="mb-1.5 block text-[13px] font-semibold text-ink">Tags</label>
            <div className="mb-2 flex flex-wrap gap-1.5">
              {tags.map((tag) => (
                <span key={tag} className="flex items-center gap-1.5 rounded-full border border-line-strong px-2.5 py-1 text-[12px] text-ink">
                  {tag}
                  <button type="button" onClick={() => setTags((prev) => prev.filter((t) => t !== tag))} className="text-ink-soft hover:text-error">
                    <X size={11} strokeWidth={2} />
                  </button>
                </span>
              ))}
            </div>
            <div className="flex gap-2">
              <Input
                placeholder="Add a tag and press Enter"
                value={tagInput}
                onChange={(e) => setTagInput(e.target.value)}
                onKeyDown={(e) => {
                  if (e.key === "Enter") {
                    e.preventDefault();
                    addTag();
                  }
                }}
                className="h-10 flex-1"
              />
              <Button type="button" size="sm" variant="secondary" onClick={addTag}>
                Add
              </Button>
            </div>
          </div>
        </div>
      </Modal>

      <Modal
        open={Boolean(deleteTarget)}
        onClose={() => setDeleteTarget(null)}
        title="Delete category?"
        footer={
          <>
            <Button variant="ghost" onClick={() => setDeleteTarget(null)}>
              Cancel
            </Button>
            <Button variant="danger" onClick={handleDelete} disabled={deleteCategory.isPending}>
              Delete
            </Button>
          </>
        }
      >
        <p className="text-[14px] text-ink-soft">
          This will remove <span className="font-semibold text-ink">{deleteTarget?.name}</span> from the taxonomy.
          Courses already assigned to it will need to be recategorized.
        </p>
      </Modal>
    </main>
  );
}

"use client";

import { useEffect, useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import { Search, CornerDownLeft } from "lucide-react";
import { ADMIN_NAV } from "../../lib/nav-config";

type FlatItem = {
  label: string;
  href: string;
  section: string;
  icon: (typeof ADMIN_NAV)[number]["items"][number]["icon"];
};

export function CommandPalette() {
  const router = useRouter();
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState("");
  const [activeIndex, setActiveIndex] = useState(0);

  const flatItems: FlatItem[] = useMemo(
    () =>
      ADMIN_NAV.flatMap((section) =>
        section.items.map((item) => ({
          label: item.label,
          href: item.href,
          section: section.title,
          icon: item.icon
        }))
      ),
    []
  );

  const filtered = useMemo(() => {
    if (!query.trim()) return flatItems;
    const q = query.trim().toLowerCase();
    return flatItems.filter(
      (item) => item.label.toLowerCase().includes(q) || item.section.toLowerCase().includes(q)
    );
  }, [flatItems, query]);

  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      const isCmdK = (e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k";
      if (isCmdK) {
        e.preventDefault();
        setOpen((v) => !v);
      }
      if (e.key === "Escape") {
        setOpen(false);
      }
    };
    window.addEventListener("keydown", handler);
    return () => window.removeEventListener("keydown", handler);
  }, []);

  useEffect(() => {
    if (open) {
      setQuery("");
      setActiveIndex(0);
    }
  }, [open]);

  useEffect(() => {
    setActiveIndex(0);
  }, [query]);

  const navigate = (href: string) => {
    router.push(href);
    setOpen(false);
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === "ArrowDown") {
      e.preventDefault();
      setActiveIndex((i) => Math.min(i + 1, filtered.length - 1));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setActiveIndex((i) => Math.max(i - 1, 0));
    } else if (e.key === "Enter") {
      e.preventDefault();
      const target = filtered[activeIndex];
      if (target) navigate(target.href);
    }
  };

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-[100] flex items-start justify-center bg-ink/40 px-4 pt-[12vh]">
      <div className="absolute inset-0" onClick={() => setOpen(false)} />
      <div className="relative w-full max-w-[560px] overflow-hidden rounded-card border border-line bg-card shadow-xl">
        <div className="flex items-center gap-2.5 border-b border-line px-4 py-3">
          <Search size={16} strokeWidth={1.75} className="shrink-0 text-ink-soft" />
          <input
            autoFocus
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            onKeyDown={handleKeyDown}
            placeholder="Jump to a page... (Users, Payouts, Settings)"
            className="h-6 w-full bg-transparent text-[14px] text-ink outline-none placeholder:text-ink-soft/60"
          />
          <kbd className="hidden shrink-0 rounded border border-line-strong px-1.5 py-0.5 text-[10px] font-medium text-ink-soft sm:block">
            Esc
          </kbd>
        </div>

        <div className="max-h-[360px] overflow-y-auto p-2">
          {filtered.length === 0 ? (
            <p className="px-3 py-6 text-center text-[13px] text-ink-soft">No matching pages.</p>
          ) : (
            filtered.map((item, index) => (
              <button
                key={item.href}
                type="button"
                onClick={() => navigate(item.href)}
                onMouseEnter={() => setActiveIndex(index)}
                className={`flex w-full items-center gap-3 rounded-control px-3 py-2.5 text-left text-[13px] transition-colors duration-100 ease-out ${
                  index === activeIndex ? "bg-accent text-paper" : "text-ink hover:bg-paper-dim"
                }`}
              >
                <item.icon size={15} strokeWidth={1.75} className="shrink-0" />
                <span className="flex-1 truncate">{item.label}</span>
                <span
                  className={`shrink-0 text-[11px] ${index === activeIndex ? "text-paper/70" : "text-ink-soft/60"}`}
                >
                  {item.section}
                </span>
                {index === activeIndex && <CornerDownLeft size={12} strokeWidth={1.75} className="shrink-0" />}
              </button>
            ))
          )}
        </div>
      </div>
    </div>
  );

export function findStaffUser(email: string, password: string): StaffUser | null {
  const match = STAFF_USERS.find(
    (u) => u.email.toLowerCase() === email.trim().toLowerCase() && u.password === password
  );
  return match ?? null;
}

// Route prefix -> roles allowed to access it. Checked in order, first match wins.
// Any route not listed here is allowed for all roles (e.g. /dashboard).
const ROUTE_PERMISSIONS: { prefix: string; roles: StaffRole[] }[] = [
  // Settings - Super Admin only
  { prefix: "/settings/roles", roles: ["super_admin"] },
  { prefix: "/settings/commission-tiers", roles: ["super_admin"] },
  { prefix: "/settings/payment-gateways", roles: ["super_admin"] },
  { prefix: "/settings/platform", roles: ["super_admin"] },
  { prefix: "/settings/legal-documents", roles: ["super_admin"] },

  // Finance - Super Admin, Admin, Finance Officer
  { prefix: "/finance", roles: ["super_admin", "admin", "finance_officer"] },

  // Analytics - Super Admin, Admin, Finance Officer
  { prefix: "/analytics", roles: ["super_admin", "admin", "finance_officer"] },

  // Courses - Super Admin, Admin only
  { prefix: "/courses", roles: ["super_admin", "admin"] },

  // Marketing - Super Admin, Admin only
  { prefix: "/marketing", roles: ["super_admin", "admin"] },

  // Audit log - Super Admin, Admin only
  { prefix: "/logs/audit", roles: ["super_admin", "admin"] },

  // Users - Super Admin, Admin, Support Officer
  { prefix: "/users", roles: ["super_admin", "admin", "support_officer"] },

  // Moderation (reviews, forum, content) - Super Admin, Admin, Support Officer
  { prefix: "/moderation", roles: ["super_admin", "admin", "support_officer"] },

  // Support tickets - Super Admin, Admin, Support Officer
  { prefix: "/support", roles: ["super_admin", "admin", "support_officer"] }

  // /dashboard and anything else not listed above -> allowed for all roles
];

export function isRouteAllowed(pathname: string, role: StaffRole): boolean {
  const rule = ROUTE_PERMISSIONS.find((r) => pathname === r.prefix || pathname.startsWith(`${r.prefix}/`));
  if (!rule) return true; // not restricted -> everyone can access
  return rule.roles.includes(role);
}

export function getAllowedRolesForRoute(prefix: string): StaffRole[] {
  const rule = ROUTE_PERMISSIONS.find((r) => prefix === r.prefix || prefix.startsWith(`${r.prefix}/`) || r.prefix.startsWith(prefix));
  if (!rule) return ["super_admin", "admin", "finance_officer", "support_officer"];
  return rule.roles;
}

// ---- Cookie helpers (client-side, no backend) ----

const SESSION_COOKIE = "saachi_admin_session";
const ROLE_COOKIE = "saachi_admin_role";
const NAME_COOKIE = "saachi_admin_name";
const EMAIL_COOKIE = "saachi_admin_email";

export function setAuthCookies(user: StaffUser) {
  const maxAge = 60 * 60 * 24 * 7; // 7 days
  document.cookie = `${SESSION_COOKIE}=demo_token; path=/; max-age=${maxAge}`;
  document.cookie = `${ROLE_COOKIE}=${user.role}; path=/; max-age=${maxAge}`;
  document.cookie = `${NAME_COOKIE}=${encodeURIComponent(user.name)}; path=/; max-age=${maxAge}`;
  document.cookie = `${EMAIL_COOKIE}=${encodeURIComponent(user.email)}; path=/; max-age=${maxAge}`;
}

export function clearAuthCookies() {
  document.cookie = `${SESSION_COOKIE}=; path=/; max-age=0`;
  document.cookie = `${ROLE_COOKIE}=; path=/; max-age=0`;
  document.cookie = `${NAME_COOKIE}=; path=/; max-age=0`;
  document.cookie = `${EMAIL_COOKIE}=; path=/; max-age=0`;
}

function readCookie(name: string): string | null {
  if (typeof document === "undefined") return null;
  const match = document.cookie.match(new RegExp(`(?:^|; )${name}=([^;]*)`));
  return match ? decodeURIComponent(match[1]) : null;
}

export function getCurrentRole(): StaffRole {
  const stored = readCookie(ROLE_COOKIE) as StaffRole | null;
  const valid: StaffRole[] = ["super_admin", "admin", "finance_officer", "support_officer"];
  return stored && valid.includes(stored) ? stored : "super_admin";
}

export function getCurrentName(): string {
  return readCookie(NAME_COOKIE) ?? "Abdul Rehman Irfan";
}

export function getCurrentEmail(): string {
  return readCookie(EMAIL_COOKIE) ?? "abdulrehman@saachi.ai";
}

export function setCurrentRole(role: StaffRole) {
  const maxAge = 60 * 60 * 24 * 7;
  document.cookie = `${ROLE_COOKIE}=${role}; path=/; max-age=${maxAge}`;
}

// ---- Module-level access check (used by dashboard widgets, quick actions, charts) ----
// Reuses the same ROUTE_PERMISSIONS matrix as the sidebar/route guard, so there is a
// single source of truth for "who can see what."
export function canAccessModule(modulePrefix: string, role: StaffRole): boolean {
  return getAllowedRolesForRoute(modulePrefix).includes(role);
}

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";

const PUBLIC_PATHS = ["/login"];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  const adminSession = request.cookies.get("saachi_admin_session");

  const isPublic = PUBLIC_PATHS.some((p) => pathname.startsWith(p));

  if (!isPublic && !adminSession) {
    const loginUrl = new URL("/login", request.url);
    return NextResponse.redirect(loginUrl);
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"]
};
