export function useReorderBanner() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({ id, direction }: { id: string; direction: "up" | "down" }) => {
      const sorted = [...BANNERS_STATE].sort((a, b) => a.order - b.order);
      const index = sorted.findIndex((b) => b.id === id);
      const swapWith = direction === "up" ? index - 1 : index + 1;
      if (swapWith < 0 || swapWith >= sorted.length) return delay({ id }, 100);
      const a = sorted[index];
      const b = sorted[swapWith];
      const aOrder = a.order;
      a.order = b.order;
      b.order = aOrder;
      BANNERS_STATE = sorted;
      return delay({ id }, 250);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: adminMarketingKeys.banners });
    }
  });
}

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";

import { DUMMY_SESSION } from "../dummy-data";

const SESSION_COOKIE = "saachi_session";
const SESSION_STORAGE_KEY = "saachi_demo_session";

function setSessionCookie() {
  if (typeof document === "undefined") return;
  document.cookie = `${SESSION_COOKIE}=demo_token; path=/; max-age=${60 * 60 * 24 * 7}`;
}

function clearSessionCookie() {
  if (typeof document === "undefined") return;
  document.cookie = `${SESSION_COOKIE}=; path=/; max-age=0`;
}

function readStoredSession(): Session | null {
  if (typeof window === "undefined") return null;
  const raw = window.localStorage.getItem(SESSION_STORAGE_KEY);
  if (!raw) return null;
  try {
    return JSON.parse(raw) as Session;
  } catch {
    return null;
  }
}

function storeSession(session: Session) {
  if (typeof window === "undefined") return;
  window.localStorage.setItem(SESSION_STORAGE_KEY, JSON.stringify(session));
}

function clearStoredSession() {
  if (typeof window === "undefined") return;
  window.localStorage.removeItem(SESSION_STORAGE_KEY);
}

function delay<T>(value: T, ms = 500): Promise<T> {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

  });
}

    }
  });

export type CourseListFilters = {
  category?: string;
  search?: string;
  priceMin?: number;
  priceMax?: number;
  level?: string;
  language?: string;
  sort?: string;
  page?: number;
};

function delay<T>(value: T, ms = 400): Promise<T> {
  return new Promise((resolve) => setTimeout(() => resolve(value), ms));
}

function toSummary(course: DummyCourse): Course {
  return {
    id: course.id,
    slug: course.slug,
    title: course.title,
    subtitle: course.subtitle,
    thumbnailUrl: course.thumbnailUrl,
    price: course.price,
    currency: course.currency,
    rating: course.rating,
    ratingCount: course.ratingCount,
    instructor: course.instructor,
    status: course.status
  };
}

export function useCourseList(filters: CourseListFilters = {}) {
  return useQuery({
    queryKey: courseKeys.list(filters),
    queryFn: async () => {
      let items = [...DUMMY_COURSES];

      if (filters.search) {
        const q = filters.search.toLowerCase();
        items = items.filter(
          (c) => c.title.toLowerCase().includes(q) || c.instructor.fullName.toLowerCase().includes(q)
        );
      }
      if (filters.category) {
        items = items.filter((c) => c.categoryId === filters.category);
      }
      if (filters.level) {
        items = items.filter((c) => c.level === filters.level);
      }
      if (filters.language) {
        items = items.filter((c) => c.language === filters.language);
      }
      if (typeof filters.priceMin === "number") {
        items = items.filter((c) => c.price >= (filters.priceMin as number));
      }
      if (typeof filters.priceMax === "number") {
        items = items.filter((c) => c.price <= (filters.priceMax as number));
      }

      switch (filters.sort) {
        case "newest":
          items = [...items].reverse();
          break;
        case "price_asc":
          items = [...items].sort((a, b) => a.price - b.price);
          break;
        case "price_desc":
          items = [...items].sort((a, b) => b.price - a.price);
          break;
        case "rating":
          items = [...items].sort((a, b) => b.rating - a.rating);
          break;
        case "popular":
        default:
          items = [...items].sort((a, b) => b.ratingCount - a.ratingCount);
      }

      const page = filters.page ?? 1;
      const pageSize = 12;
      const start = (page - 1) * pageSize;
      const pageItems = items.slice(start, start + pageSize).map(toSummary);

      return delay<Paginated<Course>>({
        items: pageItems,
        total: items.length,
        page,
        pageSize
      });
    }
  });
}

export type ApiError = {
  code: string;
  message: string;
  fieldErrors?: Record<string, string[]>;
};

export class ApiRequestError extends Error {
  code: string;
  fieldErrors?: Record<string, string[]>;
  status: number;

  constructor(status: number, error: ApiError) {
    super(error.message);
    this.name = "ApiRequestError";
    this.status = status;
    this.code = error.code;
    this.fieldErrors = error.fieldErrors;
  }
}

type RequestOptions = {
  method?: "GET" | "POST" | "PATCH" | "PUT" | "DELETE";
  body?: unknown;
  params?: Record<string, string | number | boolean | undefined>;
  signal?: AbortSignal;
};

function buildUrl(path: string, params?: RequestOptions["params"]): string {
  const base = process.env.NEXT_PUBLIC_API_URL ?? "";
  const url = new URL(path.replace(/^\//, ""), base.endsWith("/") ? base : `${base}/`);
  if (params) {
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined) url.searchParams.set(key, String(value));
    });
  }
  return url.toString();
}

export async function apiRequest<TResponse>(
  path: string,
  options: RequestOptions = {}
): Promise<TResponse> {
  const { method = "GET", body, params, signal } = options;

  const response = await fetch(buildUrl(path, params), {
    method,
    credentials: "include",
    headers: {
      "Content-Type": "application/json"
    },
    body: body ? JSON.stringify(body) : undefined,
    signal
  });

  if (!response.ok) {
    let errorPayload: ApiError = { code: "unknown_error", message: "Something went wrong" };
    try {
      errorPayload = await response.json();
    } catch {
      // response had no JSON body
    }
    throw new ApiRequestError(response.status, errorPayload);
  }

  if (response.status === 204) return undefined as TResponse;
  return response.json() as Promise<TResponse>;
}

import { create } from "zustand";

export type CartItem = {
  courseId: string;
  slug: string;
  title: string;
  thumbnailUrl?: string;
  instructorName: string;
  price: number;
  currency: string;
};

export type CouponState = {
  code: string;
  percentOff: number;
} | null;

// NOTE: demo-only coupon codes, validated client-side. Replace with a real
// Coupons API call once the backend (Doc 3) is available.
const KNOWN_COUPONS: Record<string, number> = {
  SAACHI10: 10,
  SAACHI25: 25
};

type CartState = {
  items: CartItem[];
  coupon: CouponState;
  addItem: (item: CartItem) => void;
  removeItem: (courseId: string) => void;
  clear: () => void;
  applyCoupon: (code: string) => { success: boolean; message: string };
  removeCoupon: () => void;
  isInCart: (courseId: string) => boolean;
  subtotal: () => number;
  discount: () => number;
  total: () => number;
};

export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],
      coupon: null,

      addItem: (item) => {
        const exists = get().items.some((i) => i.courseId === item.courseId);
        if (exists) return;
        set({ items: [...get().items, item] });
      },

      removeItem: (courseId) => {
        set({ items: get().items.filter((i) => i.courseId !== courseId) });
      },

      clear: () => set({ items: [], coupon: null }),

      applyCoupon: (code) => {
        const normalized = code.trim().toUpperCase();
        const percentOff = KNOWN_COUPONS[normalized];
        if (!percentOff) {
          return { success: false, message: "That coupon code isn't valid." };
        }
        set({ coupon: { code: normalized, percentOff } });
        return { success: true, message: `${percentOff}% discount applied.` };
      },

      removeCoupon: () => set({ coupon: null }),

      isInCart: (courseId) => get().items.some((i) => i.courseId === courseId),

      subtotal: () => get().items.reduce((sum, i) => sum + i.price, 0),

      discount: () => {
        const coupon = get().coupon;
        if (!coupon) return 0;
        return Math.round(get().subtotal() * (coupon.percentOff / 100) * 100) / 100;
      },

      total: () => Math.max(0, get().subtotal() - get().discount())
    }),
    { name: "saachi-cart" }
  )
);

export type EnrollmentRecord = {
  courseId: string;
  slug: string;
  title: string;
  thumbnailUrl?: string;
  instructorName: string;
  progressPercent: number;
  purchasedAt: string;
  completedLessonIds: string[];
};

type EnrollmentState = {
  enrollments: EnrollmentRecord[];
  isEnrolled: (courseId: string) => boolean;
  enroll: (records: Omit<EnrollmentRecord, "progressPercent" | "purchasedAt" | "completedLessonIds">[]) => void;
  markLessonComplete: (courseId: string, lessonId: string, totalLessons: number) => void;
  getEnrollment: (courseId: string) => EnrollmentRecord | undefined;
};

export const useEnrollmentStore = create<EnrollmentState>()(
  persist(
    (set, get) => ({
      enrollments: [],

      isEnrolled: (courseId) => get().enrollments.some((e) => e.courseId === courseId),

      getEnrollment: (courseId) => get().enrollments.find((e) => e.courseId === courseId),

      enroll: (records) => {
        const now = new Date().toISOString();
        const existingIds = new Set(get().enrollments.map((e) => e.courseId));
        const fresh = records
          .filter((r) => !existingIds.has(r.courseId))
          .map((r) => ({ ...r, progressPercent: 0, purchasedAt: now, completedLessonIds: [] as string[] }));
        if (fresh.length === 0) return;
        set({ enrollments: [...get().enrollments, ...fresh] });
      },

      markLessonComplete: (courseId, lessonId, totalLessons) => {
        set({
          enrollments: get().enrollments.map((e) => {
            if (e.courseId !== courseId) return e;
            const completedLessonIds = e.completedLessonIds.includes(lessonId)
              ? e.completedLessonIds
              : [...e.completedLessonIds, lessonId];
            const progressPercent =
              totalLessons > 0 ? Math.round((completedLessonIds.length / totalLessons) * 100) : 0;
            return { ...e, completedLessonIds, progressPercent };
          })
        });
      }
    }),
    { name: "saachi-enrollments" }
  )
);

  categoryOptions,

  value,
  onChange,
  className

    const isActive = value.categories.includes(cat);

      : [...value.categories, cat];
    onChange({ ...value, categories: nextCategories });

    <aside className={cn("flex w-full flex-col gap-8 rounded-card border border-line bg-card p-6", className)}>
      <div>
        <h4 className="mb-3 text-[13px] font-semibold uppercase tracking-[0.06em] text-ink-soft/70">

        </h4>
        <div className="flex flex-col gap-2.5">
          {categoryOptions.map((opt) => (
            <Checkbox
              key={opt.value}
              label={opt.label}
              checked={value.categories.includes(opt.value)}
              onChange={() => toggleCategory(opt.value)}
            />
          ))}
        </div>
      </div>

      <Select
        label="Skill level"
        placeholder="All levels"
        options={levelOptions}
        value={value.level ?? ""}
        onChange={(e) => onChange({ ...value, level: e.target.value || undefined })}
      />

      <Select

        value={value.language ?? ""}

  open: boolean;
  onClose: () => void;
  title?: string;
  children: ReactNode;
  footer?: ReactNode;
  size?: "md" | "lg";
};

export function Modal({ open, onClose, title, children, footer, size = "md" }: ModalProps) {
  useEffect(() => {
    if (!open) return;
    const onKeyDown = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4" role="dialog" aria-modal="true">
      <div className="absolute inset-0 bg-ink/40" onClick={onClose} />
      <div

type ButtonVariant = "primary" | "secondary" | "ghost" | "danger";
type ButtonSize = "sm" | "md" | "lg";

type ButtonProps = ButtonHTMLAttributes<HTMLButtonElement> & {

  size?: ButtonSize;

const SIZE_CLASSES: Record<ButtonSize, string> = {
  sm: "h-8 px-3 text-[12.5px]",
  md: "h-9 px-4 text-[13px]",
  lg: "h-11 px-5 text-[14px]"
};

const VARIANT_STYLE: Record<ButtonVariant, { bg: string; hoverBg: string; text: string; border: string }> = {
  primary: {
    bg: "var(--color-ink)",
    hoverBg: "var(--color-ink-soft)",
    text: "var(--color-paper)",
    border: "var(--color-ink)"
  },
  secondary: {
    bg: "var(--color-card)",
    hoverBg: "var(--color-paper-dim)",
    text: "var(--color-ink)",
    border: "var(--color-line-strong)"
  },
  ghost: {
    bg: "transparent",
    hoverBg: "var(--color-paper-dim)",
    text: "var(--color-ink-soft)",
    border: "transparent"
  },
  danger: {
    bg: "var(--color-error)",
    hoverBg: "var(--color-error)",
    text: "var(--color-paper)",
    border: "var(--color-error)"
  }
};

export const Button = forwardRef<HTMLButtonElement, ButtonProps>(function Button(
  { variant = "primary", size = "md", className = "", type = "button", style, disabled, onMouseDown, onMouseUp, ...props },
  ref
) {
  const [hover, setHover] = useState(false);
  const [pressed, setPressed] = useState(false);

  const v = VARIANT_STYLE[variant];

  const dynamicStyle: CSSProperties = {
    backgroundColor: disabled ? v.bg : hover ? v.hoverBg : v.bg,
    color: v.text,
    borderColor: v.border,
    borderWidth: 1,
    borderStyle: "solid",
    opacity: disabled ? 0.45 : variant === "danger" && hover ? 0.9 : 1,
    transform: !disabled && hover ? (pressed ? "translateY(0) scale(0.97)" : "translateY(-1px)") : "translateY(0)",
    boxShadow: !disabled && hover ? "0 4px 10px rgba(0,0,0,0.18)" : "none",
    transition: "background-color 150ms ease-out, transform 120ms ease-out, box-shadow 150ms ease-out, opacity 150ms ease-out",
    cursor: disabled ? "not-allowed" : "pointer",
    ...style
  };

  const classes = [
    "inline-flex items-center justify-center gap-1.5 rounded-control font-medium whitespace-nowrap outline-none",
    "focus-visible:ring-2 focus-visible:ring-offset-2 focus-visible:ring-ink",
    SIZE_CLASSES[size],
    className
  ]
    .filter(Boolean)
    .join(" ");

  return (
    <button
      ref={ref}
      type={type}
      disabled={disabled}
      className={classes}
      style={dynamicStyle}
      onMouseEnter={() => setHover(true)}
      onMouseLeave={() => {
        setHover(false);
        setPressed(false);
      }}
      onMouseDown={(e) => {
        setPressed(true);
        onMouseDown?.(e);
      }}
      onMouseUp={(e) => {
        setPressed(false);
        onMouseUp?.(e);
      }}
      {...props}
    />
  );
});

export function formatRelativeTime(input: string | Date): string {
  const date = typeof input === "string" ? new Date(input) : input;
  const diffMs = date.getTime() - Date.now();
  const diffMinutes = Math.round(diffMs / 60000);

  const rtf = new Intl.RelativeTimeFormat("en", { numeric: "auto" });

  if (Math.abs(diffMinutes) < 60) return rtf.format(diffMinutes, "minute");
  const diffHours = Math.round(diffMinutes / 60);
  if (Math.abs(diffHours) < 24) return rtf.format(diffHours, "hour");
  const diffDays = Math.round(diffHours / 24);
  return rtf.format(diffDays, "day");
}

export function formatDuration(totalSeconds: number): string {
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  if (hours > 0) return `${hours}h ${minutes}m`;
  return `${minutes}m`;
}
