function cx(...parts: Array<string | false | undefined>) {
  return parts.filter(Boolean).join(" ");
}

type NotificationItem = { id: string; label: string; href: string };

  { id: "n3", label: "Your Silver Boost promotion expires in 4 days", href: "/instructor/promotions" }
];

function activeGroupIdFromPathname(pathname: string | null): string {
  if (!pathname) return "overview";
  const found = INSTRUCTOR_SIDEBAR.find((group) =>
    group.items.some((item) => pathname === item.href.split("?")[0] || pathname.startsWith(item.href.split("?")[0] + "/"))
  );
  return found?.id ?? "overview";
}

export function InstructorShell({ children }: { children: ReactNode }) {
  const pathname = usePathname();
  const router = useRouter();
  const { data: session } = useSession();
  const logout = useLogout();

  const [mobileOpen, setMobileOpen] = useState(false);
  const [notifOpen, setNotifOpen] = useState(false);
  const [userMenuOpen, setUserMenuOpen] = useState(false);
  const [activeGroupId, setActiveGroupId] = useState(() => activeGroupIdFromPathname(pathname));
  const [panelCollapsed, setPanelCollapsed] = useState(false);

  const notifRef = useRef<HTMLDivElement>(null);
  const userMenuRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    setActiveGroupId(activeGroupIdFromPathname(pathname));
  }, [pathname]);

  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (notifRef.current && !notifRef.current.contains(e.target as Node)) setNotifOpen(false);
      if (userMenuRef.current && !userMenuRef.current.contains(e.target as Node)) setUserMenuOpen(false);
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  function handleLogout() {
    logout.mutate(undefined, { onSuccess: () => router.push("/login") });
  }

  const sidebarOffsetClass = panelCollapsed ? "md:ml-[128px]" : "md:ml-[352px]";

  return (

          showBrand
          collapsed={panelCollapsed}
          onCollapsedChange={setPanelCollapsed}

"use client";

import { useMemo, useState } from "react";
import { MessageSquare, ArrowBigUp, CheckCircle2, Search } from "lucide-react";
import { useCommunityStore, type CommunityQuestion } from "@saachi/store";
import { useInstructorCourseStore } from "@saachi/store";
import { Button, Input, Select, EmptyState, Badge, Textarea, cn } from "@saachi/ui";

const FILTER_OPTIONS = [
  { label: "All questions", value: "all" },
  { label: "Unanswered", value: "unanswered" },
  { label: "Answered", value: "answered" }
];

// NOTE: demo-only "current instructor" identity used when posting a reply.
// Replace with the real authenticated instructor's name once auth/session
// (useSession from @saachi/api-client) is wired through this screen.
const CURRENT_INSTRUCTOR_NAME = "Ayesha Khan";

export default function InstructorCommunityPage() {
  const questions = useCommunityStore((s) => s.questions);
  const myCourses = useInstructorCourseStore((s) => s.courses);

  const [filter, setFilter] = useState("all");
  const [search, setSearch] = useState("");
  const [courseFilter, setCourseFilter] = useState("all");

  const myCourseIds = useMemo(() => new Set(myCourses.map((c) => c.id)), [myCourses]);

  const myQuestions = useMemo(
    () => questions.filter((q) => myCourseIds.has(q.courseId)),
    [questions, myCourseIds]
  );

  const filtered = useMemo(() => {
    let items = [...myQuestions];
    if (filter === "unanswered") items = items.filter((q) => q.answers.length === 0);
    if (filter === "answered") items = items.filter((q) => q.answers.length > 0);
    if (courseFilter !== "all") items = items.filter((q) => q.courseId === courseFilter);
    if (search.trim()) {
      const s = search.toLowerCase();
      items = items.filter((q) => q.body.toLowerCase().includes(s) || q.authorName.toLowerCase().includes(s));
    }
    return items.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
  }, [myQuestions, filter, courseFilter, search]);

  const unansweredCount = myQuestions.filter((q) => q.answers.length === 0).length;

  const courseOptions = [
    { label: "All courses", value: "all" },
    ...myCourses.map((c) => ({ label: c.title, value: c.id }))
  ];

  return (
    <div>
      <div className="mb-6 flex flex-wrap items-end justify-between gap-3">
        <div>
          <h1 className="mb-1 font-display text-[26px] font-bold text-ink">Community Q&amp;A</h1>
          <p className="text-[14px] text-ink-soft">
            Answer learner questions across all of your courses.
            {unansweredCount > 0 && (
              <span className="ml-2 font-semibold text-gold-dark">{unansweredCount} awaiting a reply</span>
            )}
          </p>
        </div>
        <div className="flex flex-wrap items-center gap-2">
          <div className="relative">
            <Search size={14} className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-ink-soft" />
            <input
              value={search}
              onChange={(e) => setSearch(e.target.value)}
              placeholder="Search questions..."
              className="h-10 w-56 rounded-control border border-line-strong bg-card pl-9 pr-3 text-[13px] text-ink outline-none placeholder:text-ink-soft/50 focus:border-gold"
            />
          </div>
          <Select options={courseOptions} value={courseFilter} onChange={(e) => setCourseFilter(e.target.value)} className="w-52" />
        </div>
      </div>

      <div className="mb-6 flex gap-1.5">
        {FILTER_OPTIONS.map((opt) => (
          <button
            key={opt.value}
            type="button"
            onClick={() => setFilter(opt.value)}
            className={cn(
              "rounded-full px-3.5 py-1.5 text-[12.5px] font-semibold transition-colors duration-150 ease-out",
              filter === opt.value ? "bg-ink text-paper" : "bg-paper-dim text-ink-soft hover:text-ink"
            )}

function ModuleList({
  modules,
  onChange
}: {
  modules: InstructorCourseModule[];
  onChange: (modules: InstructorCourseModule[]) => void;
}) {
  const [dragIndex, setDragIndex] = useState<number | null>(null);
  const [expanded, setExpanded] = useState<Set<string>>(() => new Set(modules.map((m) => m.id)));

  function toggleExpanded(id: string) {
    setExpanded((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  }

  function handleDragStart(index: number) {
    setDragIndex(index);
  }

  function handleDragOver(e: DragEvent, index: number) {
    e.preventDefault();
    if (dragIndex === null || dragIndex === index) return;
    const reordered = [...modules];
    const [moved] = reordered.splice(dragIndex, 1);
    reordered.splice(index, 0, moved);
    setDragIndex(index);
    onChange(reordered);
  }

  function handleDrop() {
    setDragIndex(null);
  }

  function addModule() {
    const newModule: InstructorCourseModule = { id: nextId("mod"), title: `Module ${modules.length + 1}`, lessons: [] };
    setExpanded((prev) => new Set(prev).add(newModule.id));
    onChange([...modules, newModule]);
  }

  function updateModule(id: string, patch: Partial<InstructorCourseModule>) {
    onChange(modules.map((m) => (m.id === id ? { ...m, ...patch } : m)));
  }

  function removeModule(id: string) {
    onChange(modules.filter((m) => m.id !== id));
  }

  function addLesson(moduleId: string, type: LessonType) {
    onChange(
      modules.map((m) =>
        m.id === moduleId
          ? { ...m, lessons: [...m.lessons, { id: nextId("les"), title: `New ${type} lesson`, type }] }
          : m
      )
    );
  }

  function updateLesson(moduleId: string, lessonId: string, patch: Partial<InstructorCourseLesson>) {
    onChange(
      modules.map((m) =>
        m.id === moduleId
          ? { ...m, lessons: m.lessons.map((l) => (l.id === lessonId ? { ...l, ...patch } : l)) }
          : m
      )
    );
  }

  function removeLesson(moduleId: string, lessonId: string) {
    onChange(
      modules.map((m) => (m.id === moduleId ? { ...m, lessons: m.lessons.filter((l) => l.id !== lessonId) } : m))
    );
  }

  function reorderLesson(moduleId: string, fromIndex: number, toIndex: number) {
    onChange(
      modules.map((m) => {
        if (m.id !== moduleId) return m;
        const lessons = [...m.lessons];
        const [moved] = lessons.splice(fromIndex, 1);
        lessons.splice(toIndex, 0, moved);
        return { ...m, lessons };
      })
    );
  }

  return (
    <div className="flex flex-col gap-3">
      {modules.length === 0 ? (
        <EmptyState title="No modules yet" description="Add your first module to start building the curriculum." />
      ) : (
        modules.map((m, index) => (
          <div
            key={m.id}
            draggable
            onDragStart={() => handleDragStart(index)}
            onDragOver={(e) => handleDragOver(e, index)}
            onDrop={handleDrop}
            className={cn(
              "rounded-card border border-line bg-card transition-shadow",
              dragIndex === index && "opacity-60 shadow-lg"
            )}
          >
            <div className="flex items-center gap-2 px-4 py-3">
              <span className="cursor-grab text-ink-soft/50" title="Drag to reorder">
                <GripVertical size={16} />
              </span>
              <button
                type="button"
                onClick={() => toggleExpanded(m.id)}
                className="flex h-6 w-6 shrink-0 items-center justify-center text-ink-soft"
                aria-label="Toggle module"
              >
                <ChevronDown
                  size={15}
                  className="transition-transform duration-200"
                  style={{ transform: expanded.has(m.id) ? "rotate(0deg)" : "rotate(-90deg)" }}
                />
              </button>

              <span className="ml-auto shrink-0 text-[12px] text-ink-soft">{m.lessons.length} lesson(s)</span>
              <button
                type="button"

              >
                <Trash2 size={14} />

            {expanded.has(m.id) && (
              <div className="border-t border-line px-4 py-3">
                <LessonList

                  onReorder={(from, to) => reorderLesson(m.id, from, to)}

function QuestionCard({
  question,
  index,
  isDragging,
  onDragStart,
  onDragOver,
  onDrop,
  onChange,
  onRemove
}: {
  question: QuizQuestion;
  index: number;
  isDragging: boolean;
  onDragStart: () => void;
  onDragOver: (e: DragEvent) => void;
  onDrop: () => void;
  onChange: (patch: Partial<QuizQuestion>) => void;
  onRemove: () => void;
}) {
  const [collapsed, setCollapsed] = useState(false);

  function updateOption(optId: string, text: string) {
    onChange({
      options: question.options.map((o) => (o.id === optId ? { ...o, text } : o))
    });
  }

  function removeOption(optId: string) {
    if (question.options.length <= 2) return;
    const newOptions = question.options.filter((o) => o.id !== optId);
    const patch: Partial<QuizQuestion> = { options: newOptions };
    if (question.correctOptionId === optId) {
      patch.correctOptionId = newOptions[0]?.id ?? "";
    }
    onChange(patch);
  }

  function addOption() {
    if (question.options.length >= 6) return;
    onChange({ options: [...question.options, blankOption()] });
  }

  return (
    <div
      draggable
      onDragStart={onDragStart}
      onDragOver={onDragOver}
      onDrop={onDrop}
      className={cn(
        "rounded-card border border-line bg-card transition-shadow duration-150",
        isDragging && "opacity-60 shadow-xl"
      )}
    >
      {/* Card header */}
      <div className="flex items-center gap-2 px-4 py-3">
        <span
          className="cursor-grab text-ink-soft/40 transition-colors hover:text-ink-soft"
          title="Drag to reorder"
        >
          <GripVertical size={16} />
        </span>

        <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-full bg-gold-tint text-[11px] font-semibold text-gold-dark">
          {index + 1}
        </span>

        <p className="min-w-0 flex-1 truncate text-[13.5px] font-semibold text-ink">
          {question.questionText.trim() || <span className="text-ink-soft/50 font-normal">Untitled question</span>}
        </p>

        <button
          type="button"
          onClick={() => setCollapsed((v) => !v)}
          className="flex h-8 w-8 shrink-0 items-center justify-center rounded-control text-ink-soft hover:bg-paper-dim"
        >
          {collapsed ? <ChevronDown size={15} /> : <ChevronUp size={15} />}
        </button>

        <button
          type="button"
          onClick={onRemove}
          className="flex h-8 w-8 shrink-0 items-center justify-center rounded-control text-ink-soft transition-colors hover:bg-error-tint hover:text-error"
          title="Delete question"
        >
          <Trash2 size={14} />
        </button>
      </div>

      {/* Card body */}
      {!collapsed && (
        <div className="border-t border-line px-4 py-4 flex flex-col gap-4">
          {/* Question text */}
          <Textarea
            label="Question"
            rows={2}
            value={question.questionText}
            onChange={(e) => onChange({ questionText: e.target.value })}
            placeholder="Type the question here..."
          />

          {/* Options */}
          <div>
            <p className="mb-2 text-[12.5px] font-semibold text-ink-soft uppercase tracking-[0.05em]">
              Options <span className="normal-case font-normal text-ink-soft/60">(click circle to mark correct)</span>
            </p>
            <div className="flex flex-col gap-2">
              {question.options.map((opt) => (
                <OptionRow
                  key={opt.id}
                  option={opt}
                  isCorrect={opt.id === question.correctOptionId}
                  onChange={(text) => updateOption(opt.id, text)}
                  onRemove={() => removeOption(opt.id)}
                  onMarkCorrect={() => onChange({ correctOptionId: opt.id })}
                  canRemove={question.options.length > 2}

}

function QuizPreview({ questions }: { questions: QuizQuestion[] }) {
  const [currentIdx, setCurrentIdx] = useState(0);
  const [states, setStates] = useState<Record<string, PreviewState>>({});

  const question = questions[currentIdx];
  if (!question) {
    return (
      <EmptyState
        title="No questions yet"
        description="Add at least one question to preview the quiz."
      />
    );
  }

  const state: PreviewState = states[question.id] ?? {
    selectedOptionId: null,
    submitted: false
  };

  function selectOption(optId: string) {
    if (state.submitted) return;
    setStates((prev) => ({
      ...prev,
      [question.id]: { selectedOptionId: optId, submitted: false }
    }));
  }

  function submit() {
    if (!state.selectedOptionId) return;
    setStates((prev) => ({
      ...prev,
      [question.id]: { ...prev[question.id], submitted: true }
    }));
  }

  const isCorrect =
    state.submitted && state.selectedOptionId === question.correctOptionId;

  return (
    <div className="flex flex-col gap-4">
      {/* Progress */}
      <div className="flex items-center justify-between">
        <span className="text-[12px] text-ink-soft">
          Question {currentIdx + 1} of {questions.length}
        </span>
        <div className="flex gap-1">
          {questions.map((_, i) => (
            <button
              key={i}
              type="button"
              onClick={() => setCurrentIdx(i)}
              className={cn(
                "h-1.5 rounded-full transition-all",
                i === currentIdx ? "w-5 bg-gold-dark" : "w-1.5 bg-line-strong"
              )}
            />
          ))}
        </div>
      </div>

      {/* Question */}
      <div className="rounded-card border border-line bg-paper-dim/30 p-5">
        <p className="text-[15px] font-semibold text-ink leading-snug mb-4">
          {question.questionText || <span className="text-ink-soft/50">Untitled question</span>}
        </p>

        <div className="flex flex-col gap-2">
          {question.options.map((opt) => {
            const selected = state.selectedOptionId === opt.id;
            const correct = opt.id === question.correctOptionId;
            const showResult = state.submitted;

            return (
              <button
                key={opt.id}
                type="button"
                onClick={() => selectOption(opt.id)}
                disabled={state.submitted}
                className={cn(
                  "flex items-center gap-3 rounded-control border px-4 py-2.5 text-left text-[13.5px] transition-all duration-150 disabled:cursor-default",
                  !showResult && selected
                    ? "border-gold-dark/60 bg-gold-tint font-medium text-ink"
                    : !showResult
                    ? "border-line bg-card text-ink hover:border-gold-dark/40 hover:bg-gold-tint/30"
                    : showResult && correct
                    ? "border-teal/40 bg-teal-light font-medium text-ink"
                    : showResult && selected && !correct
                    ? "border-error/40 bg-error-tint text-ink"
                    : "border-line bg-card text-ink-soft"
                )}
              >
                {showResult && correct && (
                  <CheckCircle2 size={15} className="shrink-0 text-teal" />
                )}
                {showResult && selected && !correct && (
                  <AlertCircle size={15} className="shrink-0 text-error" />
                )}
                {!showResult && (
                  <span
                    className={cn(
                      "flex h-4 w-4 shrink-0 items-center justify-center rounded-full border-2",
                      selected ? "border-gold-dark bg-gold-dark" : "border-line-strong"
                    )}
                  >
                    {selected && <span className="h-1.5 w-1.5 rounded-full bg-white" />}
                  </span>
                )}
                <span className="flex-1">
                  {opt.text || <span className="italic text-ink-soft/50">Empty option</span>}
                </span>
              </button>
            );
          })}
        </div>

        {/* Feedback */}
        {state.submitted && (
          <div
            className={cn(
              "mt-4 rounded-control border px-4 py-3 text-[13px]",
              isCorrect
                ? "border-teal/30 bg-teal-light text-teal"
                : "border-error/30 bg-error-tint text-error"
            )}
          >
            <p className="font-semibold mb-0.5">
              {isCorrect ? "Correct!" : "Incorrect"}
            </p>
            {question.explanation && (
              <p className="text-[12.5px] text-ink-soft">{question.explanation}</p>
            )}
          </div>
        )}

        {/* Submit / Next */}
        <div className="mt-4 flex gap-2">
          {!state.submitted ? (
            <Button
              size="sm"
              disabled={!state.selectedOptionId}
              onClick={submit}
            >
              Check answer
            </Button>
          ) : currentIdx < questions.length - 1 ? (
            <Button size="sm" onClick={() => setCurrentIdx((i) => i + 1)}>
              Next question â†’
            </Button>
          ) : (
            <p className="text-[13px] text-teal font-medium flex items-center gap-1.5">
              <CheckCircle2 size={15} />
              Quiz complete
            </p>
          )}
        </div>
      </div>
    </div>
  );
}

  const [title, setTitle] = useState("Untitled quiz");
  const [description, setDescription] = useState("");
  const [questions, setQuestions] = useState<QuizQuestion[]>([blankQuestion()]);

  // Derived
  const course = useMemo(
    () => courses.find((c) => c.id === courseId),
    [courses, courseId]
  );

  const modules = course?.modules ?? [];

  const moduleOptions = useMemo(
    () => [
      { label: "Select a moduleâ€¦", value: "" },
      ...modules.map((m) => ({ label: m.title, value: m.id }))
    ],
    [modules]
  );

    () => getQuizzesForCourse(courseId),
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [courseId, quizzes]
  );

    () => [
      { label: "+ New quiz", value: "__new__" },
      ...courseQuizzes.map((q) => ({
        label: q.title,
        value: q.id
      }))
    ],
    [courseQuizzes]
  );

  useEffect(() => {
    setModuleId("");
    setQuizId("__new__");

  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [courseId]);

  // When quiz selection changes, load it or reset to blank
  useEffect(() => {
    if (quizId === "__new__") {
      resetDraft();
      return;
    }
    const found = quizzes.find((q) => q.id === quizId);
    if (found) {
      setTitle(found.title);
      setDescription(found.description);
      setModuleId(found.moduleId);

  }, [quizId]);

    setTitle("Untitled quiz");
    setDescription("");

    setQuestions((prev) => [...prev, blankQuestion()]);

    setQuestions((prev) =>
      prev.map((q, i) => (i === idx ? { ...q, ...patch } : q))
    );
  }

  function removeQuestion(idx: number) {
    setQuestions((prev) => {
      const next = prev.filter((_, i) => i !== idx);
      return next.length === 0 ? [blankQuestion()] : next;

  }

    e.preventDefault();
    if (dragIndex === null || dragIndex === targetIdx) return;
    setQuestions((prev) => {
      const next = [...prev];
      const [moved] = next.splice(dragIndex, 1);
      next.splice(targetIdx, 0, moved);
      return next;

            Build multiple-choice quizzes and attach them to your course modules.

              Saved

function RequestPayoutModal({ open, onClose }: { open: boolean; onClose: () => void }) {
  const { data: methods } = useInstructorPayoutMethods();
  const { data: balance } = useInstructorPayoutBalance();
  const requestPayout = useRequestPayout();

  const [amount, setAmount] = useState("");

    setAmount("");

  }

    setError(null);

      { amount: numeric, methodId: effectiveMethodId },
      {
        onSuccess: () => setSuccess(true),
        onError: (err) => setError(err instanceof Error ? err.message : "Something went wrong")
      }
    );

    <Modal open={open} onClose={handleClose} title={success ? "Payout requested" : "Request a payout"}>

  const [saved, setSaved] = useState(false);

function resolveGroupHref(group: InstructorNavGroup): string {
  const firstItem = group.items[0];
  if (!firstItem) return "/instructor/dashboard";
  return firstItem.href.split("?")[0];
}

export function InstructorIconRail({

  onSelect
}: {
  activeGroupId: string;
  onSelect: (id: string) => void;
}) {
  const router = useRouter();

  function handleClick(group: InstructorNavGroup) {
    onSelect(group.id);
    router.push(resolveGroupHref(group));
  }

  return (

      {INSTRUCTOR_SIDEBAR.map((group) => {

        const isActive = group.id === activeGroupId;
        return (
          <button
            key={group.id}

            className={cn(
              "relative flex h-10 w-10 shrink-0 items-center justify-center rounded-control transition-colors duration-200 ease-out",
              isActive ? "bg-gold text-ink" : "text-ink-soft hover:bg-paper-dim hover:text-ink"
            )}
            style={{ transitionTimingFunction: softSpring }}

export function InstructorDetailPanel({
  activeGroupId,
  onNavigate,
  showBrand = false,
  collapsed: collapsedProp,
  onCollapsedChange
}: {
  activeGroupId: string;
  onNavigate?: () => void;
  showBrand?: boolean;
  /** Optional controlled collapse state. If omitted, the panel manages its own state. */
  collapsed?: boolean;
  onCollapsedChange?: (collapsed: boolean) => void;
}) {
  const pathname = usePathname();
  const [expanded, setExpanded] = useState<Set<string>>(new Set());
  const [internalCollapsed, setInternalCollapsed] = useState(false);
  const collapsed = collapsedProp ?? internalCollapsed;
  const setCollapsed = (value: boolean | ((prev: boolean) => boolean)) => {
    const next = typeof value === "function" ? (value as (prev: boolean) => boolean)(collapsed) : value;
    if (onCollapsedChange) onCollapsedChange(next);
    else setInternalCollapsed(next);
  };
  const [search, setSearch] = useState("");

  const group = INSTRUCTOR_SIDEBAR.find((g) => g.id === activeGroupId) ?? INSTRUCTOR_SIDEBAR[0];

  const toggle = (key: string) => {
    setExpanded((prev) => {
      const next = new Set(prev);
      if (next.has(key)) next.delete(key);
      else next.add(key);
      return next;
    });
  };

  const filteredItems = search.trim()
    ? group.items.filter((item) => item.label.toLowerCase().includes(search.toLowerCase()))
    : group.items;

  return (
    <div
      className={cn(
        "flex h-full flex-col gap-4 border-r border-line bg-card py-4 transition-[width] duration-300 ease-out",
        collapsed ? "w-16 items-center px-0" : "w-72 px-3"
      )}
      style={{ transitionTimingFunction: softSpring }}
    >
      {showBrand && !collapsed && (
        <Link href="/instructor/dashboard" className="flex items-center gap-2 px-2 pb-2">
          <span className="font-display text-[16px] font-bold text-ink">
            Saachi<span className="text-gold-dark">AI</span>
          </span>
          <span className="rounded-full bg-paper-dim px-2 py-0.5 text-[10px] font-semibold uppercase tracking-[0.06em] text-ink-soft">
            Instructor
          </span>
        </Link>
      )}

      {/* Section title + collapse toggle */}
      <div className={cn("flex items-center", collapsed ? "justify-center" : "justify-between px-1")}>
        {!collapsed && (
          <span className="font-display text-[15px] font-semibold text-ink">{group.label}</span>
        )}
        <button
          type="button"
          aria-label={collapsed ? "Expand panel" : "Collapse panel"}
          onClick={() => setCollapsed((v) => !v)}
          className="flex h-8 w-8 items-center justify-center rounded-control text-ink-soft transition-colors duration-150 ease-out hover:bg-paper-dim hover:text-ink"
        >
          {collapsed ? <ChevronRight size={15} /> : <ChevronLeft size={15} />}
        </button>
      </div>

      {/* Search */}
      {!collapsed && (
        <div className="flex h-10 items-center gap-2 rounded-control border border-line bg-paper-dim px-3">
          <Search size={14} className="shrink-0 text-ink-soft" />
          <input
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder="Search..."
            className="w-full bg-transparent text-[13px] text-ink placeholder:text-ink-soft/60 outline-none"
          />
        </div>
      )}

      {/* Items */}
      <nav className="flex flex-1 flex-col gap-1 overflow-y-auto">
        {filteredItems.map((item, idx) => {
          const itemKey = `${group.id}-${idx}`;
          const isExpanded = expanded.has(itemKey);
          const isActive = pathname === item.href || Boolean(pathname?.startsWith(item.href.split("?")[0] + "/"));

          return (
            <div key={itemKey} className="flex flex-col">
              {item.hasChildren ? (
                <button
                  type="button"
                  onClick={() => toggle(itemKey)}
                  title={collapsed ? item.label : undefined}
                  className={cn(
                    "flex items-center gap-2 rounded-control px-2.5 py-2 text-left text-[13px] font-medium transition-colors duration-150 ease-out",
                    collapsed ? "justify-center" : "justify-between",
                    isActive ? "bg-gold text-ink" : "text-ink-soft hover:bg-paper-dim hover:text-ink"
                  )}
                >
                  {!collapsed && <span className="truncate">{item.label}</span>}
                  {!collapsed && (
                    <ChevronDown
                      size={14}
                      className="shrink-0 transition-transform duration-200 ease-out"
                      style={{
                        transitionTimingFunction: softSpring,
                        transform: isExpanded ? "rotate(180deg)" : "rotate(0deg)"
                      }}
                    />
                  )}
                  {collapsed && <span className="h-1.5 w-1.5 rounded-full bg-current" />}
                </button>
              ) : (
                <Link
                  href={item.href}
                  onClick={onNavigate}
                  title={collapsed ? item.label : undefined}
                  className={cn(
                    "flex items-center rounded-control px-2.5 py-2 text-[13px] font-medium transition-colors duration-150 ease-out",
                    collapsed ? "justify-center" : "justify-start",
                    isActive ? "bg-gold text-ink" : "text-ink-soft hover:bg-paper-dim hover:text-ink"
                  )}
                >
                  {collapsed ? <span className="h-1.5 w-1.5 rounded-full bg-current" /> : item.label}
                </Link>
              )}

              {item.hasChildren && isExpanded && !collapsed && item.children && (
                <div className="ml-2 flex flex-col gap-0.5 border-l border-line pl-2 py-1">
                  {item.children.map((child, childIdx) => (
                    <Link
                      key={`${itemKey}-${childIdx}`}
                      href={child.href}
                      onClick={onNavigate}
                      className="flex items-center justify-between rounded-control px-2.5 py-1.5 text-[12.5px] text-ink-soft transition-colors duration-100 ease-out hover:bg-paper-dim hover:text-ink"
                    >
                      <span className="truncate">{child.label}</span>
                      {child.meta && (
                        <span className="ml-2 shrink-0 rounded-full bg-paper-dim px-1.5 py-0.5 text-[10px] font-semibold text-ink-soft">
                          {child.meta}
                        </span>
                      )}
                    </Link>
                  ))}
                </div>
              )}
            </div>
          );
        })}
      </nav>
    </div>
  );
}

"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { ChevronDown, X, CircleCheck, Circle } from "lucide-react";
import { useInstructorCourseStore } from "@saachi/store";
import { Button, cn } from "@saachi/ui";

const DISMISS_KEY = "saachi-instructor-onboarding-dismissed";

type ChecklistItem = {
  id: string;
  label: string;
  description: string;
  href: string;
  done: boolean;
};

export function InstructorOnboardingChecklist() {
  const courses = useInstructorCourseStore((s) => s.courses);

  const [dismissed, setDismissed] = useState(true); // default true until we check localStorage (avoids flash)
  const [expanded, setExpanded] = useState(true);
  const [hydrated, setHydrated] = useState(false);

  useEffect(() => {
    setHydrated(true);
    const stored = window.localStorage.getItem(DISMISS_KEY);
    setDismissed(stored === "true");
  }, []);

  // NOTE: demo-only heuristics. Replace with real profile/payout/mentorship
  // status flags from the backend once available.
  const hasPublishedCourse = courses.some((c) => c.status === "published");
  const hasAnyCourse = courses.length > 0;

  const items: ChecklistItem[] = [
    {
      id: "profile",
      label: "Complete your profile",
      description: "Add a bio and portfolio link so learners know who's teaching.",
      href: "/instructor/settings",
      done: true // NOTE: demo-only, assume complete since session always has a name
    },
    {
      id: "payout",
      label: "Add a payout method",
      description: "Connect a bank or wallet account so you can withdraw earnings.",
      href: "/instructor/payouts/methods",
      done: false
    },
    {
      id: "publish",
      label: "Publish your first course",
      description: "Get your course live and start accepting enrollments.",
      href: "/instructor/courses",
      done: hasPublishedCourse
    },
    {
      id: "mentorship",
      label: "Enable mentorship",
      description: "Offer paid 1-on-1 sessions to learners who finish your courses.",
      href: "/instructor/mentorship",
      done: false
    }
  ];

  const completedCount = items.filter((i) => i.done).length;
  const allDone = completedCount === items.length;

  function handleDismiss() {
    window.localStorage.setItem(DISMISS_KEY, "true");
    setDismissed(true);
  }

  if (!hydrated || dismissed || !hasAnyCourse === false) {
    // still render normally once hydrated; this branch only prevents SSR flash
  }
  if (!hydrated) return null;
  if (dismissed) return null;

  return (
    <div className="mb-10 overflow-hidden rounded-card border border-line bg-card">
      <div className="flex items-center gap-3 px-5 py-4">
        <div className="flex-1">
          <div className="mb-1.5 flex items-center gap-2">
            <h2 className="text-[15px] font-semibold text-ink">Get set up on Saachi AI</h2>
            <span className="rounded-full bg-paper-dim px-2 py-0.5 text-[11px] font-semibold text-ink-soft">
              {completedCount} of {items.length} complete
            </span>
          </div>
          <div className="h-1.5 w-full max-w-xs overflow-hidden rounded-full bg-paper-dim">
            <div
              className="h-full rounded-full bg-gold transition-all duration-300 ease-out"
              style={{ width: `${(completedCount / items.length) * 100}%` }}
            />
          </div>
        </div>

        <button
          type="button"
          onClick={() => setExpanded((v) => !v)}
          aria-label={expanded ? "Collapse checklist" : "Expand checklist"}
          className="flex h-8 w-8 shrink-0 items-center justify-center rounded-control text-ink-soft transition-colors duration-150 ease-out hover:bg-paper-dim hover:text-ink"
        >
          <ChevronDown
            size={16}
            className="transition-transform duration-200 ease-out"
            style={{ transform: expanded ? "rotate(180deg)" : "rotate(0deg)" }}
          />
        </button>

        {allDone && (
          <button
            type="button"
            onClick={handleDismiss}
            aria-label="Dismiss checklist"
            className="flex h-8 w-8 shrink-0 items-center justify-center rounded-control text-ink-soft transition-colors duration-150 ease-out hover:bg-paper-dim hover:text-ink"
          >
            <X size={16} />
          </button>
        )}
      </div>

      {expanded && (
        <div className="border-t border-line px-5 py-2">
          {items.map((item) => (
            <div
              key={item.id}
              className="flex items-center gap-3 border-b border-line py-3 last:border-0"
            >
              <span className={cn("shrink-0", item.done ? "text-teal" : "text-ink-soft/40")}>
                {item.done ? <CircleCheck size={18} /> : <Circle size={18} />}
              </span>
              <div className="min-w-0 flex-1">
                <p
                  className={cn(
                    "text-[13.5px] font-semibold",
                    item.done ? "text-ink-soft line-through" : "text-ink"
                  )}
                >
                  {item.label}
                </p>
                <p className="text-[12px] text-ink-soft">{item.description}</p>
              </div>
              {!item.done && (
                <Link href={item.href} className="shrink-0">
                  <Button size="sm" variant="secondary">
                    Go
                  </Button>
                </Link>
              )}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

export default InstructorOnboardingChecklist;

"use client";

import { useEffect, useMemo, useRef, useState } from "react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { Search, BookOpen, Users, Star, ArrowUpRight } from "lucide-react";
import { useInstructorCourseStore } from "@saachi/store";
import { cn } from "@/lib/utils";

type ResultItem = {
  id: string;
  title: string;
  sub?: string;
  href: string;
};

type ResultGroup = {
  key: string;
  label: string;
  icon: React.ReactNode;
  items: ResultItem[];
};

// NOTE: demo-only dummy data for students/reviews. Courses come from the
// real instructor course store. Replace students/reviews with
// useInstructorStudents() / useInstructorReviews() once confirmed.
const DUMMY_STUDENTS = [
  { id: "stu_1", name: "Hina Malik", courseTitle: "Automate Everything with n8n & AI" },
  { id: "stu_2", name: "Omar Sheikh", courseTitle: "Automate Everything with n8n & AI" },
  { id: "stu_3", name: "Sara Iqbal", courseTitle: "AI Chatbots for WhatsApp Support" },
  { id: "stu_4", name: "Noor Fatima", courseTitle: "Automate Everything with n8n & AI" },
  { id: "stu_5", name: "Ali Raza", courseTitle: "AI Chatbots for WhatsApp Support" },
  { id: "stu_6", name: "Danish Iqbal", courseTitle: "Prompt Engineering for Client Work" }
];

const DUMMY_REVIEWS = [
  { id: "rev_i1", author: "Bilal S.", courseTitle: "Automate Everything with n8n & AI", comment: "Very practical, loved it." },
  { id: "rev_i2", author: "Fatima R.", courseTitle: "AI Chatbots for WhatsApp Support", comment: "Good pacing, clear examples." },
  { id: "rev_i3", author: "Omar Sheikh", courseTitle: "Automate Everything with n8n & AI", comment: "Best automation course I've taken." }
];

export function InstructorSearchBar() {
  const router = useRouter();
  const courses = useInstructorCourseStore((s) => s.courses);

  const [query, setQuery] = useState("");
  const [open, setOpen] = useState(false);
  const wrapperRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function handleClickOutside(e: MouseEvent) {
      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
        setOpen(false);
      }
    }
    document.addEventListener("mousedown", handleClickOutside);
    return () => document.removeEventListener("mousedown", handleClickOutside);
  }, []);

  const groups: ResultGroup[] = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return [];

    const courseResults: ResultItem[] = courses
      .filter((c) => c.title.toLowerCase().includes(q))
      .slice(0, 5)
      .map((c) => ({
        id: c.id,
        title: c.title,
        sub: c.status === "published" ? "Published" : c.status === "pending_review" ? "Pending review" : "Draft",
        href: `/instructor/courses/${c.id}/edit`
      }));

    const studentResults: ResultItem[] = DUMMY_STUDENTS
      .filter((s) => s.name.toLowerCase().includes(q) || s.courseTitle.toLowerCase().includes(q))
      .slice(0, 5)
      .map((s) => ({
        id: s.id,
        title: s.name,
        sub: s.courseTitle,
        href: "/instructor/students"
      }));

    const reviewResults: ResultItem[] = DUMMY_REVIEWS
      .filter(
        (r) =>
          r.author.toLowerCase().includes(q) ||
          r.courseTitle.toLowerCase().includes(q) ||
          r.comment.toLowerCase().includes(q)
      )
      .slice(0, 5)
      .map((r) => ({
        id: r.id,
        title: `${r.author} - ${r.courseTitle}`,
        sub: r.comment,
        href: "/instructor/reviews"
      }));

    const out: ResultGroup[] = [];
    if (courseResults.length > 0) out.push({ key: "courses", label: "Courses", icon: <BookOpen size={12} />, items: courseResults });
    if (studentResults.length > 0) out.push({ key: "students", label: "Students", icon: <Users size={12} />, items: studentResults });
    if (reviewResults.length > 0) out.push({ key: "reviews", label: "Reviews", icon: <Star size={12} />, items: reviewResults });
    return out;
  }, [courses, query]);

  const totalResults = groups.reduce((sum, g) => sum + g.items.length, 0);
  const showDropdown = open && query.trim().length > 0;

  function handleSelect(href: string) {
    setOpen(false);
    setQuery("");
    router.push(href);
  }

  return (
    <div ref={wrapperRef} className="relative hidden md:block">
      <div className="flex h-9 w-64 items-center gap-2 rounded-control border border-line bg-paper-dim px-3">
        <Search size={14} className="shrink-0 text-ink-soft" />
        <input
          value={query}
          onChange={(e) => {
            setQuery(e.target.value);
            setOpen(true);
          }}
          onFocus={() => setOpen(true)}
          placeholder="Search your courses, students..."
          className="w-full bg-transparent text-[12.5px] text-ink placeholder:text-ink-soft/60 outline-none"
        />
      </div>

      {showDropdown && (
        <div className="absolute right-0 top-[calc(100%+8px)] z-50 w-[340px] overflow-hidden rounded-card border border-line bg-card shadow-xl">
          {totalResults === 0 ? (
            <div className="px-4 py-8 text-center text-[12.5px] text-ink-soft">
              No results for &quot;{query}&quot;
            </div>
          ) : (
            <div className="max-h-[380px] overflow-y-auto p-1.5">
              {groups.map((group) => (
                <div key={group.key} className="mb-1 last:mb-0">
                  <div className="flex items-center gap-1.5 px-2.5 pb-1 pt-2 text-[10.5px] font-semibold uppercase tracking-[0.06em] text-ink-soft/60">
                    {group.icon}
                    {group.label}
                  </div>
                  {group.items.map((item) => (
                    <button
                      key={item.id}
                      type="button"
                      onClick={() => handleSelect(item.href)}
                      className="flex w-full items-center justify-between gap-3 rounded-control px-2.5 py-2 text-left transition-colors duration-100 ease-out hover:bg-paper-dim"
                    >
                      <div className="min-w-0">
                        <p className="truncate text-[12.5px] font-medium text-ink">{item.title}</p>
                        {item.sub && <p className="truncate text-[11px] text-ink-soft">{item.sub}</p>}
                      </div>
                      <ArrowUpRight size={12} className="shrink-0 text-ink-soft/50" />
                    </button>
                  ))}
                </div>
              ))}
            </div>
          )}
        </div>
      )}
    </div>
  );
}

export default InstructorSearchBar;
